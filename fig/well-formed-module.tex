\begin{subfigure}{\textwidth}
\flushleft \shadebox{$\Gamma; \Lambda \vdash m : \Delta; \Lambda'$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{empty}}]
{
  \strut
}
{
  \Gamma; \Lambda \vdash \varepsilon : \varnothing; \Lambda
}
\and
\inferrule*[lab={\ruleName{import}}]
{
  M' = x_1.x_2.\ldots.x_{n+1} \\
  M' \in \dom(\mathcal{M}) \\
  M' \neq M \\
  \Gamma \setminus \set{x_1}; \Lambda, x_1 : M' \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exImport{M'} \cons m : \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{from-import}}]
{
  M' \in \dom(\mathcal{M}) \\
  M' \neq M \\
  \vec{x} \importsR_{M'} \Delta'; \Lambda'' \\
  (\Gamma \setminus \Lambda'') \runion \Delta'; (\Lambda \setminus \Delta') \runion \Lambda'' \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exFromImport{M'}{\vec{x}} \cons m : \Delta' \runion \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{statement}}]
{
  \Gamma; \Lambda \vdash s : \Assigns{\Delta} \\
  \Gamma \runion \Delta; \Lambda \setminus \Delta \vdash m : \Delta'; \Lambda' \\
  \capturesS(s) \cap \assignsS(m) = \varnothing
}
{
  \Gamma; \Lambda \vdash s \cons m : \Delta \runion \Delta'; \Lambda'
}
\and
\inferrule*[lab={\ruleName{class}}]
{
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \Gamma; \Lambda \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exDataclass{C}{\phi} \cons m : \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{class-extend}}]
{
  \Lambda(B) = (M.B(\vec{y}), B') \\
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \vec{x} \cap \fields(M.B) = \varnothing \\
  \Gamma; \Lambda \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exDataclassExtend{C}{B}{\phi} \cons m : \Delta; \Lambda'
}
\end{mathpar}
\subcaption{Body $m$ of module $M$ is well-formed in $\Gamma$ and $\Lambda$}
\end{subfigure}
