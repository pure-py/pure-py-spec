  q' \in \dom(\mathcal{M}) \\
  q' \neq q \\

  \Gamma \runion \Delta' \vdash m : \Delta
}
{
  \Gamma \vdash \exFromImport{q'}{\vec{x}} \cons m : \Delta' \runion \Delta
}
\and
\inferrule*[lab={\ruleName{statement}}]

  \Gamma \vdash s : \Assigns{\Delta} \\
  \Gamma \runion \Delta \vdash m : \Delta' \\

}
{
  \Gamma \vdash s \cons m : \Delta \runion \Delta'
}

{
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \Gamma, c : (q.c(\vec{x}), \bot) \vdash m : \Delta
}
{
  \Gamma \vdash \exDataclass{c}{\phi} \cons m : \Delta
}

{
  \Gamma(c') = (q.c'(\vec{y}), c'') \\
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \vec{x} \cap \fields(q.c') = \varnothing \\
  \Gamma, c : (q.c(\vec{x}), c') \vdash m : \Delta
}
{
  \Gamma \vdash \exDataclassExtend{c}{c'}{\phi} \cons m : \Delta
}
