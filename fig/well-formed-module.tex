\flushleft \shadebox{$\Gamma \vdash m : \Delta$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{empty}}]
{
  \strut
}
{
  \Gamma \vdash \varepsilon : \varnothing
}
\and
\inferrule*[lab={\ruleName{import}}]
{
  q' = x_1.x_2.\ldots.x_{n+1} \\
  q' \in \dom(\mathcal{M}) \\
  q' \neq q \\
  \vdash_{\mathcal{M}} q' \\
  \Gamma, x_1 : \modEntry{q'} \vdash m : \Delta
}
{
  \Gamma \vdash \exImport{q'} \cons m : \Delta
}
\and
\inferrule*[lab={\ruleName{from-import}}]
{
  q' \in \dom(\mathcal{M}) \\
  q' \neq q \\
  \vdash_{\mathcal{M}} q' \\
  q', \vec{x} \importsR \Gamma' \\
  \Gamma \runion \Gamma' \vdash m : \Delta
}
{
  \Gamma \vdash \exFromImport{q'}{\vec{x}} \cons m : \Gamma' \runion \Delta
}
\and
\inferrule*[lab={\ruleName{statement}}]
{
  \Gamma \vdash s : \Assigns{\Delta} \\
  \Gamma \runion \Delta \vdash m : \Delta' \\
  \capturesS(s) \cap \assignsS(m) = \varnothing
}
{
  \Gamma \vdash s \cons m : \Delta \runion \Delta'
}
\and
\inferrule*[lab={\ruleName{class}}]
{
  \ownFields(\phi) = \vec{x} \text{ distinct} \\
  \Gamma, c : \clsEntry{\Gamma}{q}{\vec{x}}{\bot} \vdash m : \Delta
}
{
  \Gamma \vdash \exDataclass{c}{\phi} \cons m : \set{c : \clsEntry{\Gamma}{q}{\vec{x}}{\bot}} \runion \Delta
}
\and
\inferrule*[lab={\ruleName{class-extend}}]
{
  \Gamma(c') = \clsEntry{\Delta}{q}{\vec{y}}{c''} \\
  \ownFields(\phi) = \vec{x} \text{ distinct} \\
  \vec{x} \cap \fields(\Gamma(c')) = \varnothing \\
  \Gamma, c : \clsEntry{\Gamma}{q}{\vec{x}}{c'} \vdash m : \Delta
}
{
  \Gamma \vdash \exDataclassExtend{c}{c'}{\phi} \cons m : \set{c : \clsEntry{\Gamma}{q}{\vec{x}}{c'}} \runion \Delta
}
\end{mathpar}
\subcaption{Body $m$ of module $q$ is well-formed in $\Gamma$}
\end{subfigure}
