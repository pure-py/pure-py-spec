\flushleft \shadebox{$\Gamma; \Lambda \vdash m : \Delta; \Lambda'$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{empty}}]
{
  \strut
}
{
  \Gamma; \Lambda \vdash \varepsilon : \varnothing; \Lambda
}
\and
\inferrule*[lab={\ruleName{import}}]
{
  q' = x_1.x_2.\ldots.x_{n+1} \\
  q' \in \dom(\mathcal{q}) \\
  q' \neq q \\
  \vdash_{\mathcal{q}} q' \\
  \Gamma; \Lambda, x_1 : q' \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exImport{q'} \cons m : \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{from-import}}]
{
  q' \in \dom(\mathcal{q}) \\
  q' \neq q \\
  \vdash_{\mathcal{q}} q' \\
  \vec{x} \importsR_{q'} \Delta'; \Lambda'' \\
  \Gamma \runion \Delta'; \Lambda \runion \Lambda'' \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exFromImport{q'}{\vec{x}} \cons m : \Delta' \runion \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{statement}}]
{
  \Gamma; \Lambda \vdash s : \Assigns{\Delta} \\
  \Gamma \runion \Delta; \Lambda \vdash m : \Delta'; \Lambda' \\
  \capturesS(s) \cap \assignsS(m) = \varnothing
}
{
  \Gamma; \Lambda \vdash s \cons m : \Delta \runion \Delta'; \Lambda'
}
\and
\inferrule*[lab={\ruleName{class}}]
{
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \Gamma; \Lambda, c : (q.c(\vec{x}), \bot) \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exDataclass{c}{\phi} \cons m : \Delta; \Lambda'
}
\and
\inferrule*[lab={\ruleName{class-extend}}]
{
  \Lambda(c') = (q.c'(\vec{y}), c'') \\
  \decls(\phi) = \vec{x} \text{ distinct} \\
  \vec{x} \cap \fields(q.c') = \varnothing \\
  \Gamma; \Lambda, c : (q.c(\vec{x}), c') \vdash m : \Delta; \Lambda'
}
{
  \Gamma; \Lambda \vdash \exDataclassExtend{c}{c'}{\phi} \cons m : \Delta; \Lambda'
}
\end{mathpar}
\subcaption{Body $m$ of module $q$ is well-formed in $\Gamma$ and $\Lambda$}
\end{subfigure}
