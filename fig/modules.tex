\flushleft \shadebox{$\vdash_{\mathcal{M}} M$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{module}}]
{
  \Gamma_M \vdash_M \mathcal{M}(M) : \Delta \\
  \imports(\mathcal{M}(M)) \subseteq \dom(\mathcal{M})
}
{
  \vdash_{\mathcal{M}} M
}
\end{mathpar}

}
{
  \vdash \langle E, \mathcal{M} \rangle
