\begin{align*}
\eqOp(n, n) &{}= \pyinline{True} && \text{if } n \text{ is not } \pyinline{nan} \\
\eqOp(n, n') &{}= \pyinline{False} && \text{if } n \neq n' \text{ and neither is } \pyinline{nan} \\
\eqOp(w, w) &{}= \pyinline{True} \\
\eqOp(w, w') &{}= \pyinline{False} && \text{if } w \neq w' \\
\eqOp(\pyinline{None}, \pyinline{None}) &{}= \pyinline{True} \\
\eqOp(\pyinline{None}, v) = \eqOp(v, \pyinline{None}) &{}= \pyinline{False} && \text{if } v \neq \pyinline{None} \\
\eqOp(\pyinline{True}, \pyinline{True}) = \eqOp(\pyinline{False}, \pyinline{False}) &{}= \pyinline{True} \\
\eqOp(\pyinline{True}, \pyinline{False}) = \eqOp(\pyinline{False}, \pyinline{True}) &{}= \pyinline{False} \\
\\
\eqOp(\exList{\varepsilon}, \exList{\varepsilon}) = \eqOp(\exTuple{\varepsilon}, \exTuple{\varepsilon}) &{}= \pyinline{True} \\
\eqOp(\exList{v \cons \seq{v}}, \exList{v' \cons \seq{v}\,'}) &{}= \pyinline{False} && \text{if } \eqOp(v, v') = \pyinline{False} \\
\eqOp(\exList{v \cons \seq{v}}, \exList{v' \cons \seq{v}\,'}) &{}= \eqOp(\exList{\seq{v}}, \exList{\seq{v}\,'}) && \text{if } \eqOp(v, v') = \pyinline{True} \\
\eqOp(\exTuple{v \cons \seq{v}}, \exTuple{v' \cons \seq{v}\,'}) &{}= \pyinline{False} && \text{if } \eqOp(v, v') = \pyinline{False} \\
\eqOp(\exTuple{v \cons \seq{v}}, \exTuple{v' \cons \seq{v}\,'}) &{}= \eqOp(\exTuple{\seq{v}}, \exTuple{\seq{v}\,'}) && \text{if } \eqOp(v, v') = \pyinline{True} \\
\eqOp(\exList{\seq{v}}, \exList{\seq{v}\,'}) = \eqOp(\exTuple{\seq{v}}, \exTuple{\seq{v}\,'}) &{}= \pyinline{False} && \text{if } \seq{v}, \seq{v}\,' \text{ differ in length} \\
\\
\eqOp(d, d') &{}= \eqEntries(d, d') && \text{if } \dom(d) = \dom(d') \\
\eqOp(d, d') &{}= \pyinline{False} && \text{if } \dom(d) \neq \dom(d') \\
\\
\eqEntries(\varepsilon, d') &{}= \pyinline{True} \\
\eqEntries(\bind{w}{v} \cons d, d') &{}= \pyinline{False} && \text{if } \eqOp(v, d'(w)) = \pyinline{False} \\
\eqEntries(\bind{w}{v} \cons d, d') &{}= \eqEntries(d, d') && \text{if } \eqOp(v, d'(w)) = \pyinline{True} \\
\\
\eqOp(\exObj{\Gamma}{C}{\rho}, \exObj{\Gamma'}{C}{\rho'}) &{}= \textstyle\bigwedge_{x \in \fields(C)} \eqOp(\rho(x), \rho'(x)) \\
\eqOp(\exObj{\Gamma}{C}{\rho}, \exObj{\Gamma'}{C'}{\rho'}) &{}= \pyinline{False} && \text{if } C \neq C' \\
\\
\memOp(\exList{\varepsilon}, v) = \memOp(\exTuple{\varepsilon}, v) &{}= \pyinline{False} \\
\memOp(\exList{v' \cons \seq{v}}, v) &{}= \pyinline{True} && \text{if } \eqOp(v, v') = \pyinline{True} \\
\memOp(\exList{v' \cons \seq{v}}, v) &{}= \memOp(\exList{\seq{v}}, v) && \text{if } \eqOp(v, v') = \pyinline{False} \\
\memOp(\exTuple{v' \cons \seq{v}}, v) &{}= \pyinline{True} && \text{if } \eqOp(v, v') = \pyinline{True} \\
\memOp(\exTuple{v' \cons \seq{v}}, v) &{}= \memOp(\exTuple{\seq{v}}, v) && \text{if } \eqOp(v, v') = \pyinline{False} \\
\memOp(d, w) &{}= \pyinline{True} && \text{if } w \in \dom(d) \\
\memOp(d, w) &{}= \pyinline{False} && \text{if } w \notin \dom(d) \\
\memOp(w', w) &{}= \pyinline{True} && \text{if } w' = u \cons w \cons u' \text{ for some } u, u' \\
\memOp(w', w) &{}= \pyinline{False} && (\text{otherwise})
\end{align*}
