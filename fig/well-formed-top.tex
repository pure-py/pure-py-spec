\inferrule*[lab={\ruleName{empty}}]
{
  \strut
}
{
  \Gamma \vdash_M \varepsilon : \varnothing
}
\and
\inferrule*[lab={\ruleName{import}}]
{
  M' = x_1.x_2.\ldots.x_{n+1}
}
{
  \Gamma \vdash_M \exImport{M'} : \statusDefAssigned(\set{x_1})
}
\and
\inferrule*[lab={\ruleName{from-import}}]
{
  \strut
}
{
  \Gamma \vdash_M \exFromImport{M'}{x_1, \ldots, x_{n+1}} : \statusDefAssigned(\set{x_1, \ldots, x_{n+1}})
}
\and
\inferrule*[lab={\ruleName{seq}}]
{
  \Gamma \vdash_M t : \Delta \\
  \Gamma \runion \Delta \vdash_M m : \Delta' \\
  \capturesS(t) \cap \assignsS(m) = \varnothing
}
{
  \Gamma \vdash_M t \cons m : \Delta \runion \Delta'
}
\and
\inferrule*[lab={\ruleName{stmt-tail}}]
{
  \Gamma \vdash s : \Assigns{\Delta} \\
  \Gamma \runion \Delta \vdash_M m : \Delta' \\
  \capturesS(s) \cap \assignsS(m) = \varnothing
}
{
  \Gamma \vdash_M s \cons m : \Delta \runion \Delta'
}
\and
\inferrule*[lab={\ruleName{class-tail}}]
{
  x_1, \ldots, x_n \text{ distinct} \\
  \Gamma \vdash_M m : \Delta
}
{
  \Gamma \vdash_M \exDataclass{C}{\exField{x_1} \;\ldots\; \exField{x_n}} \cons m : \Delta
}
\and
\inferrule*[lab={\ruleName{class-extend-tail}}]
{
  B \in \dom(\Lambda_M) \\
  x_1, \ldots, x_n \text{ distinct and disjoint from } \fields_M(B) \\
  \Gamma \vdash_M m : \Delta
}
{
  \Gamma \vdash_M \exDataclassExtend{C}{B}{\exField{x_1} \;\ldots\; \exField{x_n}} \cons m : \Delta
}
\end{mathpar}
