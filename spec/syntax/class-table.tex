\begin{specdefinition}{Class name}
\label{def:class}
\emph{Class names} $C$ are fully-qualified names
$q.c$, where $c$ is the name under which the class is defined in module $q$. A module defines each name at most
once, so class names identify classes\speconly{, and we write ``class $C$'' for the class named $C$}. A class name is also the type of the instances of the class.
\end{specdefinition}\paperonly{ }%
\begin{specdefinition}{Class table}%
\label{def:class-table}%
A \emph{class table} $\Sigma$ maps each
class name $C$ in its domain to a \emph{class entry} $\ClassDef[\Gamma]{q}{\seq{\chi}}{i}$ (\figref{context-entries}), which records the class
definition $\chi_i$ of mutual type region $\seq{\chi}$ of module $q$ defining $C$, and the context $\Gamma$ in which
the region appears. \deleted{Every class named in $\Sigma$ is in $\dom(\Sigma)$, and the base-class relation on
$\dom(\Sigma)$ is acyclic. }\added{The base-class relation on $\dom(\Sigma)$ is acyclic, and for each $C \in \dom(\Sigma)$:}
\begin{AddedBlock}
\begin{itemize}
\item if $\base{\Sigma}(C) = D[\seq{\sigma}]$, then $D[\seq{\sigma}]$ is well-formed
(Definition~\ref{def:well-formed-type}), $D \neq \clsRange$ and $\fv(D[\seq{\sigma}]) \subseteq \typeParams{\Sigma}(C)$
\item each field type $\tau$ in $\fields{\Sigma}(C)$ is well-formed, with $\fv(\tau) \subseteq \typeParams{\Sigma}(C)$
\item $\typeParams{\Sigma}(C)$ is distinct, and field names in $\fields{\Sigma}(C)$ are distinct
\end{itemize}
\end{AddedBlock}
\end{specdefinition}
\begin{AddedBlock}
\begin{specdefinition}{Well-formed type}%
\label{def:well-formed-type}%
A type $\tau$ is \emph{well-formed} in $\Sigma$ when every class named in $\tau$ is in $\dom(\Sigma)$ and is applied to as many type arguments as it has type parameters.%
\end{specdefinition}
\end{AddedBlock}
