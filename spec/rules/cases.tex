\inferrule*[lab={\ruleName{case}}]
{
  \Sigma; \Gamma \vdash p \agrees \sigma \\
  \Sigma; \Gamma \vdash L, p \matches K, L' \binding \Delta \\
  \Sigma; \Gamma \override \Delta; \zeta \vdash s : R
}
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, \exCase{p}{s} : \Assigns{\Delta} \override R, L'
}
\and
\inferrule*[lab={\ruleName{cases}}]
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, h : S, L' \\
  \Sigma; \Gamma; \zeta; \sigma \vdash L', \seq{h} : S', L^\dagger
}
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, h \cons \seq{h} : S \mergeS S', L^\dagger
}
