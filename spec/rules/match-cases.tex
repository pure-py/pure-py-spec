\inferrule*[lab={\ruleName{match-case}}]
{
  \Sigma; \Gamma \vdash p \checks \replaced{\tau}{\sigma} \added{\binding \Delta} \\
  \Sigma; \Gamma \vdash L, p \matches K, L' \deleted{\binding \Delta} \\
  \Sigma; \Gamma \override \Delta; \zeta \vdash s : R
}
{
  \Sigma; \Gamma; \zeta\deleted{; \sigma} \vdash \added{\tau, }L, \exCase{p}{s} : \Assigns{\Delta} \override R, \added{\remaining(\Sigma, \Gamma, \tau, p), }L'
}
\and
\inferrule*[lab={\ruleName{match-cases}}]
{
  \Sigma; \Gamma; \zeta\deleted{; \sigma} \vdash \added{\tau, }L, b : S, \added{\tau', }L' \\
  \Sigma; \Gamma; \zeta\deleted{; \sigma} \vdash \added{\tau', }L', \seq{b} : S', \added{\tau^\dagger, }L^\dagger
}
{
  \Sigma; \Gamma; \zeta\deleted{; \sigma} \vdash \added{\tau, }L, b \cons \seq{b} : S \merge S', \added{\tau^\dagger, }L^\dagger
}
