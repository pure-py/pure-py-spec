\inferrule*[lab={\ruleName{match-case}}]
{
  \seqSafe(\Sigma, \Gamma, p, \sigma) \\
  \Sigma; \Gamma \vdash L, p \matches K, L' \binding \Delta \\
  \Sigma; \Gamma \override \Delta; \zeta \vdash s : R
}
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, \exCase{p}{s} : \Assigns{\Delta} \override R, L'
}
\and
\inferrule*[lab={\ruleName{match-cases}}]
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, b : S, L' \\
  \Sigma; \Gamma; \zeta; \sigma \vdash L', \seq{b} : S', L^\dagger
}
{
  \Sigma; \Gamma; \zeta; \sigma \vdash L, b \cons \seq{b} : S \merge S', L^\dagger
}
