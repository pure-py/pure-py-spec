\floorDiv(n, n') &{}= -1.0 && \quad \text{if } n' \text{ infinite and } n \text{ nonzero, of the opposite sign} \\*

\floorDiv(n, n') &{}= 0.0 && \quad \text{if } n' \text{ infinite and } n \text{ of the sign of } n' \\*
