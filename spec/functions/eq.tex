\eqOp(\ell, \ell) &{}= \pyinline{True} && \text{if } \ell \text{ is not } \pyinline{nan} \\*
\eqOp(n, n') &{}= \pyinline{False} && \text{if } n \neq n' \\*
\eqOp(w, w') &{}= \pyinline{False} && \text{if } w \neq w' \\*
\eqOp(\pyinline{None}, v) &{}= \pyinline{False} && \text{if } v \neq \pyinline{None} \\*
\eqOp(v, \pyinline{None}) &{}= \pyinline{False} && \text{if } v \neq \pyinline{None} \\*
\eqOp(\pyinline{True}, \pyinline{False}) &{}= \pyinline{False} \\*
\eqOp(\pyinline{False}, \pyinline{True}) &{}= \pyinline{False} \\*
\\
\eqOp(\exList{v_1, \ldots, v_n}, \exList{v'_1, \ldots, v'_m}) &{}= \eqElems(v_1 \cons \ldots \cons v_n, v'_1 \cons \ldots \cons v'_m) \\*
\eqOp(\exTuple{v_1, \ldots, v_n}, \exTuple{v'_1, \ldots, v'_m}) &{}= \eqElems(v_1 \cons \ldots \cons v_n, v'_1 \cons \ldots \cons v'_m) \\*
\\
\eqOp(\delta, \delta') &{}= \eqElems(v_1 \cons \ldots \cons v_n, \delta'(w_1) \cons \ldots \cons \delta'(w_n)) && \text{if } \dom(\delta) = \dom(\delta') \\*
&\qquad \text{where } \delta = \bind{w_1}{v_1} \cons \ldots \cons \bind{w_n}{v_n} \\*
\eqOp(\delta, \delta') &{}= \pyinline{False} && \text{if } \dom(\delta) \neq \dom(\delta') \\*
\\
\eqOp(\exObj{\Gamma}{C}{\rho}, \exObj{\Gamma'}{C}{\rho'}) &{}= \eqElems(\rho(x_1) \cons \ldots \cons \rho(x_n), \rho'(x_1) \cons \ldots \cons \rho'(x_n)) \\*
&\qquad \text{if } \dom(\fields(C)) = x_1, \ldots, x_n \\*
\eqOp(\exObj{\Gamma}{C}{\rho}, \exObj{\Gamma'}{D}{\rho'}) &{}= \pyinline{False} && \text{if } C \neq D
