\eqOp{\Sigma}(n, n') &{}= \val{\pyinline{True}} && \text{if } n = n' \text{ and } n \text{ is not } \pyinline{nan} \\*
\eqOp{\Sigma}(n, n') &{}= \val{\pyinline{False}} && \text{if } n \neq n' \\*
\eqOp{\Sigma}(n, n) &{}= \undefinedResult && \text{if } n \text{ is } \pyinline{nan} \\*
\eqOp{\Sigma}(w, w') &{}= \val{\pyinline{True}} && \text{if } w = w' \\*
\eqOp{\Sigma}(w, w') &{}= \val{\pyinline{False}} && \text{if } w \neq w' \\*
\eqOp{\Sigma}(\pyinline{True}, \pyinline{True}) &{}= \eqOp{\Sigma}(\pyinline{False}, \pyinline{False}) = \val{\pyinline{True}} \\*
\eqOp{\Sigma}(\pyinline{True}, \pyinline{False}) &{}= \eqOp{\Sigma}(\pyinline{False}, \pyinline{True}) = \val{\pyinline{False}} \\*
\eqOp{\Sigma}(\pyinline{None}, \pyinline{None}) &{}= \val{\pyinline{True}} \\*
\eqOp{\Sigma}(\pyinline{True}, n) &{}= \eqOp{\Sigma}(\pyinline{False}, n) = \eqOp{\Sigma}(n, \pyinline{True}) = \eqOp{\Sigma}(n, \pyinline{False}) = \undefinedResult \\*
\\
\eqOp{\Sigma}(\exList{\seq{v}}, \exList{\seq{v}\,'}) &{}= \eqElems{\Sigma}(\seq{v}, \seq{v}\,') \\*
\eqOp{\Sigma}(\exTuple{\seq{v}}, \exTuple{\seq{v}\,'}) &{}= \eqElems{\Sigma}(\seq{v}, \seq{v}\,') \\*
\\
\eqOp{\Sigma}(\delta, \delta') &{}= \eqElems{\Sigma}(v_1 \cons \ldots \cons v_n, \delta'(w_1) \cons \ldots \cons \delta'(w_n)) \\*
&\qquad \text{if } \dom(\delta) = \dom(\delta') \text{, where } \delta = \bind{w_1}{v_1} \cons \ldots \cons \bind{w_n}{v_n} \\*
\eqOp{\Sigma}(\delta, \delta') &{}= \val{\pyinline{False}} \\*
&\qquad \text{if } \dom(\delta) \neq \dom(\delta') \\*
\\
\eqOp{\Sigma}(v, v') &{}= \eqElems{\Sigma}(\iterOp(v), \iterOp(v')) \\*
&\qquad \text{if } v \text{ and } v' \text{ are values of } \clsRange \\*

&\qquad \text{if } \fields{\Sigma}(C) = \scheme{\seq{\alpha}}{\bind{\seq{x}}{\seq{\sigma}}} \text{ and } C \neq \clsRange \\*
\\
\eqOp{\Sigma}(v, v') &{}= \undefinedResult \\*
&\qquad \text{if } v \text{ and } v' \text{ are closures or primitives} \\*
