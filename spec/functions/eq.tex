\eqOp{\Sigma}(n, n') &{}= \pyinline{False} && \text{if } n \neq n' \\*
\eqOp{\Sigma}(w, w') &{}= \pyinline{False} && \text{if } w \neq w' \\*
\eqOp{\Sigma}(\pyinline{None}, v) &{}= \pyinline{False} && \text{if } v \neq \pyinline{None} \\*
\eqOp{\Sigma}(v, \pyinline{None}) &{}= \pyinline{False} && \text{if } v \neq \pyinline{None} \\*
\eqOp{\Sigma}(\pyinline{True}, \pyinline{False}) &{}= \pyinline{False} \\*
\eqOp{\Sigma}(\pyinline{False}, \pyinline{True}) &{}= \pyinline{False} \\*
\\

\\

&\qquad \text{if } \dom(\delta) \neq \dom(\delta') \\*
