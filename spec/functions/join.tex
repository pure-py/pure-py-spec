\joinTy(\Sigma, \sigma, \tau) &{}= \tau && \text{if } \Sigma \vdash \sigma \subty \tau \\*
\joinTy(\Sigma, \sigma, \tau) &{}= \sigma && \text{if } \Sigma \vdash \tau \subty \sigma \text{ and } \Sigma \nvdash \sigma \tyEquiv \tau \\*
\joinTy(\Sigma, \sigma, \tau) &{}= \tyUnion{\sigma}{\tau} && \text{otherwise} \\*[2mm]
\joinTy(\Sigma, \varepsilon) &{}= \tyNever \\*
\joinTy(\Sigma, \sigma \cons \seq{\tau}) &{}= \joinTy(\Sigma, \sigma, \joinTy(\Sigma, \seq{\tau}))
