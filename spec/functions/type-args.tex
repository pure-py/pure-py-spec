\typeArgs(\Sigma, \alpha, \tau) &{}= \set{\bind{\alpha}{\baseType(\tau)}} \\*
\typeArgs(\Sigma, \sigma, \tyUnion{\tau}{\tau'}) &{}= \joinTy{\Sigma}{\typeArgs(\Sigma, \sigma, \tau)}{\typeArgs(\Sigma, \sigma, \tau')} \\*
\typeArgs(\Sigma, \tyList{\sigma}, \tyList{\tau}) &{}= \typeArgs(\Sigma, \sigma, \tau) \\*
\typeArgs(\Sigma, \tyDict{\sigma}, \tyDict{\tau}) &{}= \typeArgs(\Sigma, \sigma, \tau) \\*
\typeArgs(\Sigma, \tyTuple{\seq{\sigma}}, \tyTuple{\seq{\tau}}) &{}= \typeArgs(\Sigma, \seq{\sigma}, \seq{\tau}) \\*
\typeArgs(\Sigma, \tyCallable{\seq{\sigma}}{\sigma}, \tyCallable{\seq{\tau}}{\tau}) &{}= \typeArgs(\Sigma, \sigma \cons \seq{\sigma}, \tau \cons \seq{\tau}) \\*
\typeArgs(\Sigma, C[\seq{\sigma}], \tau) &{}= \typeArgs(\Sigma, \seq{\sigma}, \seq{\tau}') \\*
&\qquad \rlap{$\text{if } \Sigma \vdash C[\seq{\tau}'] \instanceAt \tau$} \\*
\typeArgs(\Sigma, \tyUnion{\sigma}{\sigma'}, \tau) &{}= \varnothing \\*
&\qquad \rlap{$\text{if } \Sigma \vdash \tau \subty \sigma \text{ or } \Sigma \vdash \tau \subty \sigma'$} \\*
\typeArgs(\Sigma, \tyUnion{\sigma}{\sigma'}, \tau) &{}= \joinTy{\Sigma}{\typeArgs(\Sigma, \sigma, \tau)}{\typeArgs(\Sigma, \sigma', \tau)} \\*
&\qquad \rlap{$\text{otherwise}$} \\*
\typeArgs(\Sigma, \sigma, \tau) &{}= \varnothing \qquad \text{otherwise} \\*[2mm]
\typeArgs(\Sigma, \varepsilon, \varepsilon) &{}= \varnothing \\*
\typeArgs(\Sigma, \sigma \cons \seq{\sigma}, \tau \cons \seq{\tau}) &{}= \joinTy{\Sigma}{\typeArgs(\Sigma, \sigma, \tau)}{\typeArgs(\Sigma, \seq{\sigma}, \seq{\tau})} \\*[2mm]
(\joinTy{\Sigma}{m}{m'})(\alpha) &{}= \joinTy{\Sigma}{m(\alpha)}{m'(\alpha)} \qquad \text{if } \alpha \in \dom(m) \cap \dom(m') \\*
(\joinTy{\Sigma}{m}{m'})(\alpha) &{}= m(\alpha) \qquad \text{if } \alpha \in \dom(m) \setminus \dom(m') \\*
(\joinTy{\Sigma}{m}{m'})(\alpha) &{}= m'(\alpha) \qquad \text{if } \alpha \in \dom(m') \setminus \dom(m)
