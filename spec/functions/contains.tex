\contains(\exList{\seq{v}}, v) &{}= \containsElems(\seq{v}, v) \\*
\contains(\exTuple{\seq{v}}, v) &{}= \containsElems(\seq{v}, v) \\*
\\
\contains(\delta, w) &{}= \pyinline{True} && \text{if } w \in \dom(\delta) \\*
\contains(\delta, w) &{}= \pyinline{False} && \text{if } w \notin \dom(\delta) \\*
\\
\contains(w', w) &{}= \pyinline{True} && \text{if } w' = w_1 \concat w \concat w_2 \text{ for some } w_1, w_2 \\*
\contains(w', w) &{}= \pyinline{False} && \text{otherwise}
