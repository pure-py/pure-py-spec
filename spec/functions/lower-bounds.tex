\lowerBounds(\Sigma, \tyList{\sigma}, \tyList{\tau}) &{}= \lowerBounds(\Sigma, \sigma, \tau) \\*
\lowerBounds(\Sigma, \tyDict{\sigma}, \tyDict{\tau}) &{}= \lowerBounds(\Sigma, \sigma, \tau) \\*
\lowerBounds(\Sigma, \tyTuple{\seq{\sigma}}, \tyTuple{\seq{\tau}}) &{}= \lowerBounds(\Sigma, \seq{\sigma}, \seq{\tau}) \\*
\lowerBounds(\Sigma, \tyCallable{\seq{\sigma}}{\sigma}, \tyCallable{\seq{\tau}}{\tau}) &{}= \lowerBounds(\Sigma, \sigma \cons \seq{\sigma}, \tau \cons \seq{\tau}) \\*
\lowerBounds(\Sigma, C[\seq{\sigma}], \tau) &{}= \lowerBounds(\Sigma, \seq{\sigma}, \seq{\tau}') \\*
&\qquad \rlap{$\text{if } \Sigma \vdash C[\seq{\tau}'] \instanceAt \tau$} \\*

&\qquad \rlap{$\text{and } \tau' \text{ is the join of the disjuncts of } \tau \text{ not below the}$} \\*
&\qquad \rlap{$\text{join of the other disjuncts of } \tyUnion{\sigma}{\sigma'}$} \\*
