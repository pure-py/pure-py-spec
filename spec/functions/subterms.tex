\subterms(\tyUnion{\sigma}{\tau}) &{}= \subterms(\sigma) \cup \subterms(\tau) \\*
\subterms(\tyList{\tau}) &{}= \set{\tyList{\tau}} \cup \subterms(\tau) \\*
\subterms(\tyDict{\tau}) &{}= \set{\tyDict{\tau}} \cup \subterms(\tau) \\*
\subterms(\tyTuple{\seq{\tau}}) &{}= \set{\tyTuple{\seq{\tau}}} \cup \subterms(\seq{\tau}) \\*
\subterms(\tyCallable{\seq{\tau}}{\tau}) &{}= \set{\tyCallable{\seq{\tau}}{\tau}} \cup \subterms(\seq{\tau}) \cup \subterms(\tau) \\*
\subterms(C[\seq{\tau}]) &{}= \set{C[\seq{\tau}]} \cup \subterms(\seq{\tau}) \\*
\subterms(\tau) &{}= \set{\tau} && \text{otherwise} \\*
\subterms(\seq{\tau}) &{}= \subterms(\tau_1) \cup \cdots \cup \subterms(\tau_n)
