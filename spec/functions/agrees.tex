\agrees(p, \tyUnion{\sigma}{\tau}) &{}= \agrees(p, \sigma) \meet \agrees(p, \tau) \\*
\agrees(\patTuple{p_1, \ldots, p_n}, \tyTuple{\tau_1, \ldots, \tau_n}) &{}= \agrees(p_1, \tau_1) \meet \ldots \meet \agrees(p_n, \tau_n) \\*
\agrees(\patTuple{\seq{p}}, \tyList{\tau}) &{}= \text{False} \\*
\agrees(\patList{p_1, \ldots, p_n}, \tyList{\tau}) &{}= \agrees(p_1, \tau) \meet \ldots \meet \agrees(p_n, \tau) \\*
\agrees(\patList{\seq{p}}, \tyTuple{\seq{\tau}}) &{}= \text{False} \\*
\agrees(\patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}}, \tyDict{\tau}) &{}= \agrees(p_1, \tau) \meet \ldots \meet \agrees(p_n, \tau) \\*
\agrees(\patConstr{q}{a}, \tau) &{}= \agrees(p_1, \sigma_1) \meet \ldots \meet \agrees(p_n, \sigma_n) \\*
&\qquad \text{if } \Gamma \vdash q : C, \; \fields(C) = \bind{\seq{x}}{\seq{\sigma}}, \; p_i = \fieldMap(C, a)(x_i) \quad (\forall i) \\*
\agrees(p\;\pyinline{as}\;x, \tau) &{}= \agrees(p, \tau) \\*
\agrees(p, \tau) &{}= \text{True} && \text{otherwise}
