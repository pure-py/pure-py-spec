\matchArgs(\alpha, \sigma) &{}= \set{\bind{\alpha}{\sigma}} \\*
\matchArgs(\tyUnion{\rho}{\rho'}, \sigma) &{}= \varnothing \\*
\matchArgs(\rho, \alpha) &{}= \undetermined \\*
\matchArgs(\rho, \tyUnion{\sigma}{\sigma'}) &{}= \undetermined \\*
\matchArgs(\tyList{\rho}, \tyList{\sigma}) &{}= \matchArgs(\rho, \sigma) \\*
\matchArgs(\tyDict{\rho}, \tyDict{\sigma}) &{}= \matchArgs(\rho, \sigma) \\*
\matchArgs(\tyTuple{\seq{\rho}}, \tyTuple{\seq{\sigma}}) &{}= \matchArgs(\seq{\rho}, \seq{\sigma}) \\*
\matchArgs(\tyCallable{\seq{\rho}}{\rho}, \tyCallable{\seq{\sigma}}{\sigma}) &{}= \matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) \\*
\matchArgs(C[\seq{\rho}], C[\seq{\sigma}]) &{}= \matchArgs(\seq{\rho}, \seq{\sigma}) \\*
\matchArgs(\rho, \rho) &{}= \varnothing \\*
\matchArgs(\rho, \sigma) &{}= \clash \quad \text{(otherwise)} \clausegap
\matchArgs(\varepsilon, \varepsilon) &{}= \varnothing \\*
\matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) &{}= \gamma \override \gamma' \\*
&\qquad \rlap{$\text{if } \matchArgs(\rho, \sigma) = \gamma$} \\*
&\qquad \rlap{$\text{and } \matchArgs(\seq{\rho}, \seq{\sigma}) = \gamma'$} \\*
\matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) &{}= \clash \\*
&\qquad \rlap{$\text{if } \matchArgs(\rho, \sigma) = \clash$} \\*
&\qquad \rlap{$\text{or } \matchArgs(\seq{\rho}, \seq{\sigma}) = \clash$} \\*
\matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) &{}= \undetermined \quad \text{(otherwise)} \\*
\matchArgs(\seq{\rho}, \seq{\sigma}) &{}= \clash \quad \text{(otherwise)}
