\matchArgs(\alpha, \sigma) &{}= \set{\bind{\alpha}{\sigma}} \\*
\matchArgs(\tyUnion{\rho}{\rho'}, \sigma) &{}= \varnothing \\*
\matchArgs(\tyList{\rho}, \tyList{\sigma}) &{}= \matchArgs(\rho, \sigma) \\*
\matchArgs(\tyDict{\rho}, \tyDict{\sigma}) &{}= \matchArgs(\rho, \sigma) \\*
\matchArgs(\tyTuple{\seq{\rho}}, \tyTuple{\seq{\sigma}}) &{}= \matchArgs(\seq{\rho}, \seq{\sigma}) \\*
\matchArgs(\tyCallable{\seq{\rho}}{\rho}, \tyCallable{\seq{\sigma}}{\sigma}) &{}= \matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) \\*
\matchArgs(C[\seq{\rho}], C[\seq{\sigma}]) &{}= \matchArgs(\seq{\rho}, \seq{\sigma}) \\*
\matchArgs(\rho, \rho) &{}= \varnothing \clausegap
\matchArgs(\varepsilon, \varepsilon) &{}= \varnothing \\*
\matchArgs(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) &{}= \matchArgs(\rho, \sigma) \override \matchArgs(\seq{\rho}, \seq{\sigma})
