\matchOp{\Sigma}(\rho, \ell, v) &{}= \matchOk{\varnothing} && \text{if } \eqOp{\Sigma}(\ell, v) = \val{\pyinline{True}} \\*
\matchOp{\Sigma}(\rho, \ell, v) &{}= \noMatch && \text{if } \eqOp{\Sigma}(\ell, v) = \val{\pyinline{False}} \\*
\matchOp{\Sigma}(\rho, \ell, v) &{}= \undefinedResult && \text{if } \eqOp{\Sigma}(\ell, v) = \undefinedResult \\*
\matchOp{\Sigma}(\rho, x, v) &{}= \matchOk{\set{\bind{x}{v}}} \\*
\matchOp{\Sigma}(\rho, \patWild, v) &{}= \matchOk{\varnothing} \\*
\matchOp{\Sigma}(\rho, p\;\pyinline{as}\;x, v) &{}= \matchOp{\Sigma}(\rho, p, v) \override \matchOk{\set{\bind{x}{v}}} \\*
