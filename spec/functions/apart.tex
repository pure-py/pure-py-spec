\apart(\tyList{\rho}, \tyList{\sigma}) &{}= \apart(\rho, \sigma) \\*
\apart(\tyDict{\rho}, \tyDict{\sigma}) &{}= \apart(\rho, \sigma) \\*
\apart(\tyTuple{\seq{\rho}}, \tyTuple{\seq{\sigma}}) &{}= \apart(\seq{\rho}, \seq{\sigma}) \\*
\apart(\tyCallable{\seq{\rho}}{\rho}, \tyCallable{\seq{\sigma}}{\sigma}) &{}= \apart(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) \\*
\apart(C[\seq{\rho}], C[\seq{\sigma}]) &{}= \apart(\seq{\rho}, \seq{\sigma}) \\*
\apart(\rho, \sigma) &{}= \rho \neq \sigma \quad \text{if neither } \rho \text{ nor } \sigma \text{ is a type variable or union} \\*
\apart(\rho, \sigma) &{}= \mathsf{false} \quad \text{(otherwise)} \clausegap
\apart(\rho \cons \seq{\rho}, \sigma \cons \seq{\sigma}) &{}= \apart(\rho, \sigma) \lor \apart(\seq{\rho}, \seq{\sigma}) \\*
\apart(\varepsilon, \varepsilon) &{}= \mathsf{false} \\*
\apart(\seq{\rho}, \seq{\sigma}) &{}= \mathsf{true} \quad \text{(otherwise)}
