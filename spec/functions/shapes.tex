\shapes(\Sigma, \tyUnion{\sigma}{\tau}, H) &{}= \shapes(\Sigma, \sigma, H_\sigma) \cup \shapes(\Sigma, \tau, H_\tau) \\*
&\qquad \rlap{$\text{where } H_\sigma = \set{h \in H \mid \Sigma \vdash h : \sigma} \text{ and } H_\tau = \set{h \in H \mid \Sigma \vdash h : \tau}$} \\*
\shapes(\Sigma, \tyLiteral{\ell}, H) &{}= \varnothing && \text{if } \ell \in H \\*
\shapes(\Sigma, \tyBool, H) &{}= \varnothing && \text{if } \set{\pyinline{True}, \pyinline{False}} \subseteq H \\*
\shapes(\Sigma, \tyNone, H) &{}= \varnothing && \text{if } \pyinline{None} \in H \\*
\shapes(\Sigma, C[\seq{\tau}], H) &{}= \varnothing && \text{if } D \in \ancestors(\Sigma, C) \text{ for some } D \in H \\*
\shapes(\Sigma, \tyNever, H) &{}= \varnothing \\*
\shapes(\Sigma, \tyDict{\tau}, H) &{}= \set{\shapeDict{\tau}{\varnothing}{H}} \\*
\shapes(\Sigma, \tau, H) &{}= \set{\shapeAny{\tau}{H}} && \text{otherwise} \\*[2mm]
\shapes(\Sigma, \seq{\tau}) &{}= \shapes(\Sigma, \tau_1, \varnothing) \times \cdots \times \shapes(\Sigma, \tau_n, \varnothing)
