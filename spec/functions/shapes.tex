\shapesOf(\Sigma, \tyUnion{\sigma}{\tau}, N) &{}= \shapesOf(\Sigma, \sigma, N) \cup \shapesOf(\Sigma, \tau, N) \\*
\shapesOf(\Sigma, \tyLiteral{\ell}, N) &{}= \varnothing && \text{if } \ell \in N \\*
\shapesOf(\Sigma, \tyBool, N) &{}= \varnothing && \text{if } \set{\pyinline{True}, \pyinline{False}} \subseteq N \\*
\shapesOf(\Sigma, \tyNone, N) &{}= \varnothing && \text{if } \pyinline{None} \in N \\*
\shapesOf(\Sigma, C, N) &{}= \varnothing && \text{if } \Sigma \vdash C \subty D \text{ for some } D \in N \\*
\shapesOf(\Sigma, \tyNever, N) &{}= \varnothing \\*
\shapesOf(\Sigma, \tyDict{\tau}, N) &{}= \set{\shapeDict{\tau}{\varnothing}{N}} \\*
\shapesOf(\Sigma, \tau, N) &{}= \set{\shapeAny{\tau}{N}} && \text{otherwise} \\*[2mm]
\shapesOf(\Sigma, \seq{\tau}) &{}= \shapesOf(\Sigma, \tau_1, \varnothing) \times \cdots \times \shapesOf(\Sigma, \tau_n, \varnothing)
