\shapesOf(\Sigma, \tyUnion{\sigma}{\tau}, H) &{}= \shapesOf(\Sigma, \sigma, H) \cup \shapesOf(\Sigma, \tau, H) \\*
\shapesOf(\Sigma, \tyLiteral{\ell}, H) &{}= \varnothing && \text{if } \ell \in H \\*
\shapesOf(\Sigma, \tyBool, H) &{}= \varnothing && \text{if } \set{\pyinline{True}, \pyinline{False}} \subseteq H \\*
\shapesOf(\Sigma, \tyNone, H) &{}= \varnothing && \text{if } \pyinline{None} \in H \\*
\shapesOf(\Sigma, C, H) &{}= \varnothing && \text{if } \Sigma \vdash C \subty D \text{ for some } D \in H \\*
\shapesOf(\Sigma, \tyNever, H) &{}= \varnothing \\*
\shapesOf(\Sigma, \tyDict{\tau}, H) &{}= \set{\shapeDict{\tau}{\varnothing}{H}} \\*
\shapesOf(\Sigma, \tau, H) &{}= \set{\shapeAny{\tau}{H}} && \text{otherwise} \\*[2mm]
\shapesOf(\Sigma, \seq{\tau}) &{}= \shapesOf(\Sigma, \tau_1, \varnothing) \times \cdots \times \shapesOf(\Sigma, \tau_n, \varnothing)
