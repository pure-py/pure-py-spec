\modOp(n, n') &{}= \text{a NaN} && \quad \text{if } \NaN(n), \NaN(n') \text{ or } n \text{ infinite} \\*
\modOp(n, n') &{}= n && \quad \text{if } n' \text{ infinite and } n \text{ zero or of the sign of } n' \\*
\modOp(n, n') &{}= n' && \quad \text{if } n' \text{ infinite and } n \text{ nonzero, of the opposite sign} \\*
\modOp(n, n') &{}= \fl(n - (n' \times \lfloor n / n' \rfloor)) && \quad \text{otherwise}
