\begin{theorem}[Type soundness]
\label{thm:type-soundness}
\replaced{If $\vdash \mathcal{M}$ and $\mathcal{M}, \mathcal{A} \Rightarrow \seq{v}, \kappa$ then
$\kappa \notin \set{\errType, \errAttr, \errName, \errExcluded}$.}{No run of a well-formed program aborts with $\errType$, $\errAttr$, $\errName$ or $\errExcluded$. }
\end{theorem}
