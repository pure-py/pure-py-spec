\begin{theorem}[Type soundness]
\label{thm:type-soundness}
If $\vdash \mathcal{M}$ and $\mathcal{M}, \mathcal{A} \Rightarrow \seq{v}, \kappa$ then
\replaced{$\kappa$ is not $\errType$, $\errAttr$, $\errName$ or $\errExcluded$}{$\kappa \notin \set{\errType, \errAttr, \errName, \errExcluded}$ }.
\end{theorem}
