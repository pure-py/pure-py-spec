\inferrule*[lab={\ruleName{ty-name}}]
{ \tau \text{ a primitive type} \\ \tau \in \dom(\Gamma) }
{ \Gamma \vdash \tau }
\and
\inferrule*[lab={\ruleName{ty-class}}]
{ \Gamma(q) \text{ a class entry} }
{ \Gamma \vdash q }
\and
\inferrule*[lab={\ruleName{ty-literal}}]
{ \pyinline{Literal} \in \dom(\Gamma) }
{ \Gamma \vdash \tyLiteral{\ell} }
\and
\inferrule*[lab={\ruleName{ty-list}}]
{ \pyinline{list} \in \dom(\Gamma) \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyList{\tau} }
\and
\inferrule*[lab={\ruleName{ty-dict}}]
{ \pyinline{dict}, \pyinline{str} \in \dom(\Gamma) \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyDict{\tau} }
\and
\inferrule*[lab={\ruleName{ty-tuple}}]
{ \pyinline{tuple} \in \dom(\Gamma) \\ \Gamma \vdash \tau_i \quad (\forall i) }
{ \Gamma \vdash \tyTuple{\seq{\tau}} }
\and
\inferrule*[lab={\ruleName{ty-callable}}]
{ \pyinline{Callable} \in \dom(\Gamma) \\ \Gamma \vdash \sigma_i \quad (\forall i) \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyCallable{\seq{\sigma}}{\tau} }
\and
\inferrule*[lab={\ruleName{ty-union}}]
{ \Gamma \vdash \sigma \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyUnion{\sigma}{\tau} }
