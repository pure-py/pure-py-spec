\inferrule*[lab={\ruleName{ty-prim}}]
{ \tau \text{ primitive or literal type} }
{ \Gamma \vdash \tau }
\and
\inferrule*[lab={\ruleName{ty-class}}]
{ \Gamma(q) \text{ a class entry} }
{ \Gamma \vdash q }
\and
\inferrule*[lab={\ruleName{ty-list}}]
{ \Gamma \vdash \tau }
{ \Gamma \vdash \tyList{\tau} }
\and
\inferrule*[lab={\ruleName{ty-dict}}]
{ \Gamma \vdash \tau }
{ \Gamma \vdash \tyDict{\tau} }
\and
\inferrule*[lab={\ruleName{ty-tuple}}]
{ \Gamma \vdash \tau_i \quad (\forall i) }
{ \Gamma \vdash \tyTuple{\seq{\tau}} }
\and
\inferrule*[lab={\ruleName{ty-callable}}]
{ \Gamma \vdash \sigma_i \quad (\forall i) \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyCallable{\seq{\sigma}}{\tau} }
\and
\inferrule*[lab={\ruleName{ty-union}}]
{ \Gamma \vdash \sigma \\ \Gamma \vdash \tau }
{ \Gamma \vdash \tyUnion{\sigma}{\tau} }
