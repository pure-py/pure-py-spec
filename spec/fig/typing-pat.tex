\flushleft \shadebox{$\Gamma \vdash p \checks \tau \binding \Delta$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{pat-var}}]
{ \strut }
{ \Gamma \vdash x \checks \tau \binding \set{\bind{x}{\tau}} }
\and
\inferrule*[lab={\ruleName{pat-wild}}]
{ \strut }
{ \Gamma \vdash \patWild \checks \tau \binding \varnothing }
\and
\inferrule*[lab={\ruleName{pat-lit}}]
{ \tyLiteral{\ell} \subty \tau }
{ \Gamma \vdash \ell \checks \tau \binding \varnothing }
\and
\inferrule*[lab={\ruleName{pat-neg-lit}}]
{ \widen(n) \subty \tau }
{ \Gamma \vdash -n \checks \tau \binding \varnothing }
\and
\inferrule*[lab={\ruleName{pat-list}}]
{ \Gamma \vdash p_i \checks \tau \binding \Delta_i \quad (\forall i) }
{ \Gamma \vdash \patList{\seq{p}} \checks \tyList{\tau} \binding \Delta_1 \override \ldots \override \Delta_n }
\and
\inferrule*[lab={\ruleName{pat-tuple}}]
{ \Gamma \vdash p_i \checks \tau_i \binding \Delta_i \quad (\forall i) }
{ \Gamma \vdash \patTuple{\seq{p}} \checks \tyTuple{\tau_1, \ldots, \tau_n} \binding \Delta_1 \override \ldots \override \Delta_n }
\and
\inferrule*[lab={\ruleName{pat-dict}}]
{ \Gamma \vdash p_i \checks \tau \binding \Delta_i \quad (\forall i) }
{ \Gamma \vdash \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} \checks \tyDict{\tau} \binding \Delta_1 \override \ldots \override \Delta_n }
\and
\inferrule*[lab={\ruleName{pat-constr}}]
{
  \Gamma \vdash q : C \\
  q \subty \tau \\
  \Gamma \vdash \fieldMap(C, \seq{p})(x) \checks C(x) \binding \Delta_x \quad (\forall x \in \fields(C))
}
{ \Gamma \vdash \patConstr{q}{\seq{p}} \checks \tau \binding \textstyle\bigotimes_{x \in \fields(C)} \Delta_x }
\and
\inferrule*[lab={\ruleName{pat-as}}]
{ \Gamma \vdash p \checks \tau \binding \Delta }
{ \Gamma \vdash p\;\pyinline{as}\;x \checks \tau \binding \Delta \override \set{\bind{x}{\tau}} }
\end{mathpar}
