\small
\inferrule*[lab={\ruleName{val-lit}}]
{
  \strut
}
{
  \vdash \ell : \tyLiteral{\ell}
}
\and
\inferrule*[lab={\ruleName{val-list}}]
{
  \vdash v_i : \tau \quad (\forall i)
}
{
  \vdash \exList{v_1, \ldots, v_n} : \tyList{\tau}
}
\and
\inferrule*[lab={\ruleName{val-tuple}}]
{
  \vdash v_i : \tau_i \quad (\forall i)
}
{
  \vdash \exTuple{v_1, \ldots, v_n} : \tyTuple{\seq{\tau}}
}
\and
\inferrule*[lab={\ruleName{val-dict}}]
{
  \vdash v_i : \tau \quad (\forall i) \\
  w_1, \ldots, w_n \text{ distinct}
}
{
  \vdash \exDict{\bind{w_1}{v_1} \cons \ldots \cons \bind{w_n}{v_n}} : \tyDict{\tau}
}
\and
\inferrule*[lab={\ruleName{val-lambda}}]
{
  \vdash \rho : \Gamma \\
  \Gamma \override \set{\bind{\seq{x}}{\seq{\sigma}}} \vdash e \checks \tau
}
{
  \vdash \exLamClosure[\Gamma]{\rho}{\seq{x}}{e} : \tyCallable{\seq{\sigma}}{\tau}
}
\and
\inferrule*[lab={\ruleName{val-def}}]
{
  \vdash \rho : \Gamma \\
  \Gamma \vdash \exDef{f_1}{\bind{\seq{x}_1}{\seq{\psi}_1}}{\exRetAnnot{\psi_1}}{b_1} \cons \ldots \cons \exDef{f_{n+1}}{\bind{\seq{x}_{n+1}}{\seq{\psi}_{n+1}}}{\exRetAnnot{\psi_{n+1}}}{b_{n+1}} : \Assigns{\Delta}
}
{
  \vdash \exDefClosure[\Gamma]{\rho}{\exDef{f_1}{\bind{\seq{x}_1}{\seq{\psi}_1}}{\exRetAnnot{\psi_1}}{b_1} \cons \ldots \cons \exDef{f_{n+1}}{\bind{\seq{x}_{n+1}}{\seq{\psi}_{n+1}}}{\exRetAnnot{\psi_{n+1}}}{b_{n+1}}}{i} : \Delta(f_i)
}
\and
\inferrule*[lab={\ruleName{val-obj}}]
{
  \vdash \rho(x) : \tau \quad (\forall \bind{x}{\tau} \in \fields(C)) \\
  \vdash \rho : \Gamma
}
{
  \vdash \exObj{\Gamma}{C}{\rho} : C
}
\and
\inferrule*[lab={\ruleName{val-prim}}]
{
  \mathcal{P}(q) = (\Gamma, \rho) \\
  \rho(x) = \primVal{\primFn} \\
  \Gamma(x) = \theta
}
{
  \vdash \primVal{\primFn} : \theta
}
\and
\inferrule*[lab={\ruleName{val-opaque}}]
{
  \strut
}
{
  \vdash \opaqueVal{q.x} : \predefName
}
\and
\inferrule*[lab={\ruleName{val-mod}}]
{
  q \in \dom(\mathcal{M}) \cup \dom(\mathcal{P})
}
{
  \vdash \modStub{q} : \modStub{q}
}
\and
\inferrule*[lab={\ruleName{val-mod-loaded}}]
{
  \vdash \rho : \Gamma
}
{
  \vdash \modLoaded{q}{\rho} : \modLoaded{q}{\Gamma}
}
\and
\inferrule*[lab={\ruleName{val-class}}]
{
  \strut
}
{
  \vdash \clsEntry{\Gamma}{q.c}{\seq{x}}{\bot} : \Type{\clsEntry{\Gamma}{q.c}{\seq{x}}{\bot}}
}

\inferrule*[lab={\ruleName{val-class-extend}}]

}
{
  \vdash \clsEntry{\Gamma}{q.c}{\seq{x}}{c'} : \Type{\clsEntry{\Gamma}{q.c}{\seq{x}}{c'}}
}
