\inferrule*[lab={\ruleName{val-dict}}]
{
  \vdash v_i : \tau \quad (\forall i) \\

}
{
  \vdash \exDict{\bind{\seq{w}}{\seq{v}}} : \tyDict{\tau}
}
\and
\inferrule*[lab={\ruleName{val-lambda}}]
{
  \vdash \rho : \Gamma \\
  \Gamma \override \set{\bind{\seq{x}}{\seq{\sigma}}} \vdash e \checks \tau
}
{
  \vdash \exLamClosure[\Gamma]{\rho}{\seq{x}}{e} : \tyCallable{\seq{\sigma}}{\tau}
}
\and
\inferrule*[lab={\ruleName{val-def}}]
{
  \vdash \rho : \Gamma \\
  d_i = \exDef{f_i}{\bind{\seq{x}}{\seq{\psi}}}{\exRetAnnot{\psi}}{s} \\
  \Gamma \vdash \seq{d} : \Assigns{\Delta}
}
{
  \vdash \exDefClosure[\Gamma]{\rho}{\seq{d}}{i} : \Delta(f_i)
}
\and

  \vdash \rho : \Gamma
}
{
  \vdash \exObj{\Gamma}{C}{\rho} : C
}
\and

}
{
  \vdash \primVal{\primFn} : \theta
}
