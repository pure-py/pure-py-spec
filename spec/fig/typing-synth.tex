{
  \Gamma \vdash e_i \checks \tyStr \quad (\forall i) \\

  \Gamma \vdash e_i' \synth \tau_i \quad (\forall i \in I) \\
  \tau = \joinTy(\tau_i \mid i \in I) \\
  \Gamma \vdash e_j' \checks \tau \quad (\forall j \notin I)

  (\capturesE(e) \cup \capturesE(e')) \cap \dom(\Delta) = \varnothing
}
{
  \Gamma \vdash \exDictComp{e}{e'}{g_1\;\ldots\;g_n} \synth \tyDict{\tau}
