\flushleft \shadebox{$\Gamma \vdash e \synth \tau$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{var}}]
{
  \bind{x}{\tau} \in \Gamma
}
{
  \Gamma \vdash x \synth \tau
}
\and
\inferrule*[lab={\ruleName{literal}}]
{
  \strut
}
{
  \Gamma \vdash \ell \synth \tyLiteral{\ell}
}
\and
\inferrule*[lab={\ruleName{attr-module}}]
{
  \Gamma \vdash \exAttr{e}{x} : \tau
}
{
  \Gamma \vdash \exAttr{e}{x} \synth \tau
}
\and
\inferrule*[lab={\ruleName{attr-object}}]
{
  \Gamma \vdash e \synth q \\
  \Gamma \vdash q : C \\
  \bind{x}{\tau} \in \fields(C)
}
{
  \Gamma \vdash \exAttr{e}{x} \synth \tau
}
\and
\inferrule*[lab={\ruleName{call}}]
{
  \Gamma \vdash e \synth \tyCallable{\tau_1, \ldots, \tau_n}{\tau} \\
  \Gamma \vdash e_i \checks \tau_i \quad (\forall i)
}
{
  \Gamma \vdash \exCall{e}{e_1, \ldots, e_n} \synth \tau
}
\and
\inferrule*[lab={\ruleName{constr}}]
{
  \Gamma \vdash q : C \\
  \Gamma \vdash \fieldMap(C, \seq{e})(x) \checks \tau \quad (\forall \bind{x}{\tau} \in \fields(C))
}
{
  \Gamma \vdash \exConstr{q}{\seq{e}} \synth q
}
\and
\inferrule*[lab={\ruleName{binop}}]
{
  \odot : (\sigma, \sigma') \to \tau \\
  \Gamma \vdash e \checks \sigma \\
  \Gamma \vdash e' \checks \sigma'
}
{
  \Gamma \vdash \exBinOp{e}{\odot}{e'} \synth \tau
}
\and
\inferrule*[lab={\ruleName{unop}}]
{
  \ominus : \sigma \to \tau \\
  \Gamma \vdash e \checks \sigma
}
{
  \Gamma \vdash \exUnOp{\ominus}{e} \synth \tau
}
\and
\inferrule*[lab={\ruleName{and}}]
{
  \Gamma \vdash e \checks \tyBool \\
  \Gamma \vdash e' \checks \tyBool
}
{
  \Gamma \vdash \exAnd{e}{e'} \synth \tyBool
}
\and
\inferrule*[lab={\ruleName{or}}]
{
  \Gamma \vdash e \checks \tyBool \\
  \Gamma \vdash e' \checks \tyBool
}
{
  \Gamma \vdash \exOr{e}{e'} \synth \tyBool
}
\and
\inferrule*[lab={\ruleName{subscript-list}}]
{
  \Gamma \vdash e \synth \tyList{\tau} \\
  \Gamma \vdash e' \checks \tyInt
}
{
  \Gamma \vdash \exSubscript{e}{e'} \synth \tau
}
\and
\inferrule*[lab={\ruleName{subscript-str}}]
{
  \Gamma \vdash e \synth \tyStr \\
  \Gamma \vdash e' \checks \tyInt
}
{
  \Gamma \vdash \exSubscript{e}{e'} \synth \tyStr
}
\and
\inferrule*[lab={\ruleName{subscript-dict}}]
{
  \Gamma \vdash e \synth \tyDict{\tau} \\
  \Gamma \vdash e' \checks \tyStr
}
{
  \Gamma \vdash \exSubscript{e}{e'} \synth \tau
}
\and
\inferrule*[lab={\ruleName{subscript-tuple}}]
{
  \Gamma \vdash e \synth \tyTuple{\tau_0, \ldots, \tau_m} \\
  \Gamma \vdash e' \synth \tyLiteral{n} \\
  0 \leq n \leq m
}
{
  \Gamma \vdash \exSubscript{e}{e'} \synth \tau_n
}
\and
\inferrule*[lab={\ruleName{subscript-tuple-int}}]
{
  \Gamma \vdash e \synth \tyTuple{\tau_0, \ldots, \tau_m} \\
  \Gamma \vdash e' \synth \tyInt
}
{
  \Gamma \vdash \exSubscript{e}{e'} \synth \tau_0 \mathbin{|} \ldots \mathbin{|} \tau_m
}
