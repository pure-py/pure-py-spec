% Rows of the auxiliary-functions index; the auxindex environment supplies the table.
\auxgroup{Functions on classes and modules}
\auxrow{$\submods(q)$}{context}{stub for each submodule of $q$}{aux:submods}
\auxrow{$\uniqueTo(q', q)$}{sequence of module names}{\replaced{modules which strictly contain $q'$ and do not contain $q$}{proper prefixes of $q'$ not prefixes of $q$}, shortest first}{aux:unique-to}
\auxrow{\added{$\loads(q)$}}{\added{sequence of module names}}{\added{modules loaded by $q$, with duplicates}}{aux:loads}
\auxrow{\added{$\loadOrder(\mathcal{M})$}}{\added{sequence of module names}}{\added{modules of the program in load order}}{aux:load-order}
\auxrow{\added{$\importEnv{\mathcal{L}}(\seq{\iota})$}}{\added{environment}}{\added{environment bound by imports $\seq{\iota}$ given loaded module environments $\mathcal{L}$}}{aux:import-env}
\auxrow{$\ownFields(\phi)$, $\ownFields(\chi)$}{sequence of bindings}{fields declared by field list $\phi$ or class definition $\chi$, with their type expressions}{aux:own-fields}
\auxrow{$\typeParams{\Sigma}(C)$}{sequence of type parameters}{type parameters of class $C$}{aux:type-params}
\auxrow{$\base{\Sigma}(C)$}{instantiated class or $\bot$}{base class of $C$ at its declared type arguments}{aux:base}
\auxrow{$\fields{\Sigma}(C)$}{field scheme}{fields of class $C$ with their types, inherited first, under its type parameters}{aux:fields}
\auxrow{$\ancestors{\Sigma}(C)$}{set of classes}{$C$ and its base classes}{aux:ancestors}
\auxrow{$\fieldMap{\Sigma}(C, \seq{u})$}{map from variables}{argument supplied for each field of $C$; defined iff arguments cover them exactly}{aux:field-map}
\auxrow{$\resolve(\rho, q)$}{value}{value denoted by name $q$ in $\rho$}{aux:resolve}
\auxrow{$\nameOf(q)$}{string}{string literal naming module $q$}{aux:name}
\noalign{\vskip 2mm}
\auxgroup{Functions on values}
\auxrow{$\eqOp{\Sigma}(v, v')$}{result}{structural equality; $\aborts{\errUndef}$ between functions, between Boolean and number, and between two \pyinline{nan} values in container}{aux:eq}
\auxrow{$\eqElems{\Sigma}(\seq{v}, \seq{v}\,')$}{result}{elementwise equality of value sequences}{aux:eq-elems}
\auxrow{$\inOp{\Sigma}(v, v')$}{result}{whether $v$ is element of $v'$; $\errExcluded$ for range or non-string key, $\errType$ otherwise}{aux:in}
\auxrow{$\inElems{\Sigma}(v, \seq{v})$}{result}{whether $\seq{v}$ has element equal to $v$}{aux:in-elems}
\auxrow{$\elems(v)$}{sequence of values}{elements of list, tuple or string}{aux:elems}
\auxrow{$\iterOp(v)$}{sequence of values}{sequence drawn from $v$ by generator; keys if $v$ is dictionary, $m, \ldots, n - 1$ if $v$ is range with start $m$ and stop $n$}{aux:iter}
\auxrow{$\getitem(v, v')$}{result}{subscripting; $\errIndex$ for index out of range, $\errKey$ for missing key, $\errExcluded$ for range, Boolean index or non-string key, $\errType$ otherwise}{aux:getitem}
\auxrow{$\result(o)$}{result}{statement outcome as result: returned value, \pyinline{None} if statement completes without returning, abort unchanged}{aux:result}
\auxrow{$\matchOp{\Sigma}(\rho, p, v)$}{match result}{bindings of pattern variables if $p$ matches $v$, $\noMatch$ if not, $\errUndef$ if equality of numeric literal pattern $p$ and $v$ undefined, $\errExcluded$ for list or tuple pattern against the other or a range, $\matchFields{\Sigma}$ for constructor pattern}{aux:match}
\auxrow{$\matchFields{\Sigma}(\rho, C, a, \bind{\seq{z}}{\seq{v}})$}{match result}{sub-patterns of $a$ against object fields; $\errType$ for excess sub-patterns, $\errExcluded$ if fields not covered}{aux:match-fields}
\auxrow{$\dispatch(\rho, v, \seq{p}, \seq{s})$}{bindings and statement}{case taken by \pyinline{match}, or termination kind if matching aborts}{aux:dispatch}
\auxrow{$\update(\delta, \delta')$}{dictionary entries}{$\delta$ updated by $\delta'$ in order, later replacing earlier}{aux:update}
\auxrow{$\setitem(\delta, w, v)$}{dictionary entries}{$\delta$ with $w$ bound to $v$, replacing earlier binding}{aux:setitem}
