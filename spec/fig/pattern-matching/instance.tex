\flushleft \shadebox{$\Sigma \vdash C[\seq{\sigma}] \instanceAt \tau$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{instance-above}}]
{
  \Sigma \vdash \tau \subty C[\seq{\sigma}] \\

}

  \Sigma \vdash C[\seq{\sigma}] \instanceAt \tau
}

}

  \Sigma \vdash C[] \instanceAt \tyObject
}

\inferrule*[lab={\ruleName{instance-below}}]
{
  \typeParams{\Sigma}(C) = \seq{\alpha} \\
  \ancestorArgs{\Sigma}(C[\seq{\alpha}], D) = \seq{\tau} \\
  \matchArgs(\seq{\tau}, \seq{\sigma}') = \gamma \\
  \gamma(\alpha_i) = \sigma_i \quad (\forall i) \\
  \Sigma \vdash C[\seq{\sigma}] \subty D[\seq{\sigma}']
}
{
  \Sigma \vdash C[\seq{\sigma}] \instanceAt D[\seq{\sigma}']
}

\begin{subfigure}{\textwidth}
\flushleft \shadebox{$\tau \mathrel{\#} \sigma$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{apart-list}}]
{
  \tau \mathrel{\#} \sigma
}
{
  \tyList{\tau} \mathrel{\#} \tyList{\sigma}
}
\and
\inferrule*[lab={\ruleName{apart-dict}}]
{
  \tau \mathrel{\#} \sigma
}
{
  \tyDict{\tau} \mathrel{\#} \tyDict{\sigma}
}
\and
\inferrule*[lab={\ruleName{apart-tuple}}]
{
  \seq{\tau} \mathrel{\#} \seq{\sigma}
}
{
  \tyTuple{\seq{\tau}} \mathrel{\#} \tyTuple{\seq{\sigma}}
}
\and
\inferrule*[lab={\ruleName{apart-callable}}]
{
  (\tau \cons \seq{\tau}) \mathrel{\#} (\sigma \cons \seq{\sigma})
}
{
  \tyCallable{\seq{\tau}}{\tau} \mathrel{\#} \tyCallable{\seq{\sigma}}{\sigma}
}
\and
\inferrule*[lab={\ruleName{apart-class}}]
{
  \seq{\tau} \mathrel{\#} \seq{\sigma}
}
{
  C[\seq{\tau}] \mathrel{\#} C[\seq{\sigma}]
}
\and
\inferrule*[lab={\ruleName{apart-literal}}]
{
  \ell \neq \ell'
}
{
  \tyLiteral{\ell} \mathrel{\#} \tyLiteral{\ell'}
}
\and
\inferrule*[lab={\ruleName{apart-differ}}]
{
  \text{neither } \tau \text{ nor } \sigma \text{ a type variable or union} \\
  \tau, \sigma \text{ with different outermost type constructors}
}
{
  \tau \mathrel{\#} \sigma
}
\end{mathpar}
\flushleft \shadebox{$\seq{\tau} \mathrel{\#} \seq{\sigma}$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{apart-head}}]
{
  \tau \mathrel{\#} \sigma
}
{
  (\tau \cons \seq{\tau}) \mathrel{\#} (\sigma \cons \seq{\sigma})
}
\and
\inferrule*[lab={\ruleName{apart-tail}}]
{
  \seq{\tau} \mathrel{\#} \seq{\sigma}
}
{
  (\tau \cons \seq{\tau}) \mathrel{\#} (\sigma \cons \seq{\sigma})
}
\and
\inferrule*[lab={\ruleName{apart-length}}]
{
  |\seq{\tau}| \neq |\seq{\sigma}|
}
{
  \seq{\tau} \mathrel{\#} \seq{\sigma}
}
\end{mathpar}
\subcaption{Types $\tau$ and $\sigma$, or sequences $\seq{\tau}$ and $\seq{\sigma}$, are apart}
\label{fig:apart}
\end{subfigure}
