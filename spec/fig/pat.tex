\inferrule*[lab={\ruleName{pat-literal}}]

  \strut

{
  \ell \notin N \\
  \Gamma \vdash \tyLiteral{\ell} \subty \tau

\inferrule*[lab={\ruleName{pat-rest-constr}}]
{
  \Gamma \vdash q : C \\
  C \notin N \\
  \tau, C \text{ comparable} \\
  \fields(C) = \bind{\seq{x}}{\seq{\sigma}} \\
  p_i = \fieldMap(C, a)(x_i) \quad (\forall i) \\
  \Gamma \vdash (\shapeAny{\sigma_1}{\varnothing}, \ldots, \shapeAny{\sigma_n}{\varnothing}), \seq{p} \matches K, L \binding \Delta
}
{
  \Gamma \vdash \shapeAny{\tau}{N}, \patConstr{q}{a} \matches \set{\patConstr{C}{\seq{m}} \mid \seq{m} \in K}, \set{\patConstr{C}{\seq{k}} \mid \seq{k} \in L} \cup \shapesOf(\tau, N \cup \set{C}) \binding \Delta
}
\and
\inferrule*[lab={\ruleName{pat-rest-tuple}}]
{
  \Gamma \vdash (\shapeAny{\tau_1}{\varnothing}, \ldots, \shapeAny{\tau_n}{\varnothing}), \seq{p} \matches K, L \binding \Delta
}
{
  \Gamma \vdash \shapeAny{\tyTuple{\tau_1, \ldots, \tau_n}}{N}, \patTuple{p_1, \ldots, p_n} \matches \set{\patTuple{\seq{m}} \mid \seq{m} \in K}, \set{\patTuple{\seq{k}} \mid \seq{k} \in L} \binding \Delta
}

\inferrule*[lab={\ruleName{pat-rest-list}}]
{
  n \notin N \\
  \Gamma \vdash (\shapeAny{\tau}{\varnothing}, \ldots, \shapeAny{\tau}{\varnothing}), p_1 \cons \ldots \cons p_n \matches K, L \binding \Delta

  \Gamma \vdash \shapeAny{\tyList{\tau}}{N}, \patList{p_1, \ldots, p_n} \matches \set{\shapeList{\tau}{\seq{m}} \mid \seq{m} \in K}, \set{\shapeList{\tau}{\seq{k}} \mid \seq{k} \in L} \cup \shapesOf(\tyList{\tau}, N \cup \set{n}) \binding \Delta
}
\and
\inferrule*[lab={\ruleName{pat-constr}}]
{
  \Gamma \vdash q' : C' \\
  \Gamma \vdash C \subty C' \\
  \fields(C') = \bind{\seq{x}}{\seq{\sigma}} \\
  p_i = \fieldMap(C', a)(x_i) \quad (\forall i \leq n) \\
  p_i = \patWild \quad (\forall i > n) \\
  \Gamma \vdash \seq{k}, \seq{p} \matches K, L \binding \Delta
}
{
  \Gamma \vdash \patConstr{C}{\seq{k}}, \patConstr{q'}{a} \matches \set{\patConstr{C}{\seq{m}} \mid \seq{m} \in K}, \set{\patConstr{C}{\seq{k}\,'} \mid \seq{k}\,' \in L} \binding \Delta
}

\inferrule*[lab={\ruleName{pat-tuple}}]

}
\and
\inferrule*[lab={\ruleName{pat-list}}]
{
  \Gamma \vdash \seq{k}, \seq{p} \matches K, L \binding \Delta
}

  \Gamma \vdash \shapeList{\tau}{k_1, \ldots, k_n}, \patList{p_1, \ldots, p_n} \matches \set{\shapeList{\tau}{\seq{m}} \mid \seq{m} \in K}, \set{\shapeList{\tau}{\seq{k}\,'} \mid \seq{k}\,' \in L} \binding \Delta
}

\inferrule*[lab={\ruleName{pat-dict-nil}}]
{
  \strut

{
  \beta(w_1) = k \\
  w_1 \notin \set{w_2, \ldots, w_{n+1}} \\
  \Gamma \vdash k, p_1 \matches K, L \binding \Delta \\
  \Gamma \vdash \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{m}}}{N} \mid m \in K}, \patDict{\bind{w_2}{p_2}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \matches K', L' \binding \Delta' \\
  L^\dagger = \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{k'}}}{N} \mid k' \in L} \cup L'
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \matches K', L^\dagger \binding \Delta \disjunion \Delta'
}

\inferrule*[lab={\ruleName{pat-dict-unbound}}]

  \Gamma \vdash \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{m}}}{N} \mid m \in K}, \patDict{\bind{w_2}{p_2}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \matches K', L' \binding \Delta' \\
  L^\dagger = \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{k}}}{N} \mid k \in L} \cup L' \cup \set{\shapeDict{\tau}{\beta}{N \cup \set{w_1}}}
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \matches K', L^\dagger \binding \Delta \disjunion \Delta'

{
  \Gamma \vdash k_i, p_i \matches K_i, L_i \binding \Delta_i \quad (\forall i) \\

  L = \textstyle\bigcup_i \set{(m_1, \ldots, m_{i-1}, k\,', k_{i+1}, \ldots, k_n) \mid k\,' \in L_i, \; m_j \in K_j \text{ for } j < i}
}

  \Gamma \vdash (k_1, \ldots, k_n), (p_1, \ldots, p_n) \matches K, L \binding \Delta_1 \disjunion \ldots \disjunion \Delta_n
}

{
  \varnothing \neq M = \set{k \in L \mid \Gamma \vdash k, p \matches K_k, L_k \binding \Delta_k} \\
  \Delta = \set{\bind{x}{\joinTy(\Delta_k(x) \mid k \in M)} \mid x \in \binds(p)}
}
{
  \Gamma \vdash L, p \matches \textstyle\bigcup_{k \in M} K_k, \; \textstyle\bigcup_{k \in M} L_k \cup (L \setminus M) \binding \Delta
}
