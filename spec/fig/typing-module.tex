{
  x_1, \ldots, x_{n+1} \text{ distinct} \\
  \Sigma \vdash_{\mathsf{mod}} q' : \Delta \binding \Sigma_0 \\
  q^\dagger_1 \cons \cdots \cons q^\dagger_m = \uniqueTo(q', q) \\
  \Sigma_{j-1} \vdash_{\mathsf{mod}} q^\dagger_j : \Delta_j \binding \Sigma_j \quad (\forall j) \\

  \Sigma_{m+n+1}; q \vdash \seq{\iota} : \Gamma \binding \Sigma'
}
{
  \Sigma; q \vdash \exFromImport{q'}{x_1, \ldots, x_{n+1}} \cons \seq{\iota} : \set{\bind{x_1}{\theta_1}, \ldots, \bind{x_{n+1}}{\theta_{n+1}}} \extend \Gamma \binding \Sigma'
