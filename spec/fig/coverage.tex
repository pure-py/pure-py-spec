\flushleft \shadebox{$\Gamma \vdash k, p \splitTo K, L \binding \Delta$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{split-var}}]
{
  \strut
}
{
  \Gamma \vdash k, x \splitTo \set{k}, \varnothing \binding \set{\bind{x}{\shapeTy(k)}}
}
\and
\inferrule*[lab={\ruleName{split-wild}}]
{
  \strut
}
{
  \Gamma \vdash k, \patWild \splitTo \set{k}, \varnothing \binding \varnothing
}
\and
\inferrule*[lab={\ruleName{split-as}}]
{
  \Gamma \vdash k, p \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash k, p\;\pyinline{as}\;x \splitTo K, L \binding \Delta \override \set{\bind{x}{\joinTy(\shapeTy(m) \mid m \in K)}}
}
\and
\inferrule*[lab={\ruleName{split-literal}}]
{
  \strut
}
{
  \Gamma \vdash \ell, \ell \splitTo \set{\ell}, \varnothing \binding \varnothing
}
\and
\inferrule*[lab={\ruleName{split-rest-literal}}]
{
  \ell \notin N \\
  \Gamma \vdash \tyLiteral{\ell} \subty \tau
}
{
  \Gamma \vdash \shapeAny{\tau}{N}, \ell \splitTo \set{\ell}, \shapesOf(\tau, N \cup \set{\ell}) \binding \varnothing
}
\and
\inferrule*[lab={\ruleName{split-rest-constr}}]
{
  q \notin N \\
  \tau, q \text{ comparable} \\
  \Gamma \vdash q : C \\
  \fields(C) = \bind{\seq{x}}{\seq{\sigma}} \\
  p_i = \fieldMap(C, a)(x_i) \quad (\forall i) \\
  \Gamma \vdash (\shapeAny{\sigma_1}{\varnothing}, \ldots, \shapeAny{\sigma_n}{\varnothing}), \seq{p} \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \shapeAny{\tau}{N}, \patConstr{q}{a} \splitTo \set{\patConstr{q}{\seq{m}} \mid \seq{m} \in K}, \set{\patConstr{q}{\seq{k}} \mid \seq{k} \in L} \cup \shapesOf(\tau, N \cup \set{q}) \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-rest-tuple}}]
{
  \Gamma \vdash (\shapeAny{\tau_1}{\varnothing}, \ldots, \shapeAny{\tau_n}{\varnothing}), \seq{p} \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \shapeAny{\tyTuple{\tau_1, \ldots, \tau_n}}{N}, \patTuple{p_1, \ldots, p_n} \splitTo \set{\patTuple{\seq{m}} \mid \seq{m} \in K}, \set{\patTuple{\seq{k}} \mid \seq{k} \in L} \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-rest-list}}]
{
  n \notin N \\
  \Gamma \vdash (\shapeAny{\tau}{\varnothing}, \ldots, \shapeAny{\tau}{\varnothing}), p_1 \cons \ldots \cons p_n \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \shapeAny{\tyList{\tau}}{N}, \patList{p_1, \ldots, p_n} \splitTo \set{\shapeList{\tau}{\seq{m}} \mid \seq{m} \in K}, \set{\shapeList{\tau}{\seq{k}} \mid \seq{k} \in L} \cup \shapesOf(\tyList{\tau}, N \cup \set{n}) \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-constr}}]
{
  \Gamma \vdash q \subty q' \\
  \Gamma \vdash q' : C \\
  \fields(C) = \bind{\seq{x}}{\seq{\sigma}} \\
  p_i = \fieldMap(C, a)(x_i) \quad (\forall i \leq n) \\
  p_i = \patWild \quad (\forall i > n) \\
  \Gamma \vdash \seq{k}, \seq{p} \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \patConstr{q}{\seq{k}}, \patConstr{q'}{a} \splitTo \set{\patConstr{q}{\seq{m}} \mid \seq{m} \in K}, \set{\patConstr{q}{\seq{k}\,'} \mid \seq{k}\,' \in L} \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-tuple}}]
{
  \Gamma \vdash \seq{k}, \seq{p} \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \patTuple{k_1, \ldots, k_n}, \patTuple{p_1, \ldots, p_n} \splitTo \set{\patTuple{\seq{m}} \mid \seq{m} \in K}, \set{\patTuple{\seq{k}\,'} \mid \seq{k}\,' \in L} \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-list}}]
{
  \Gamma \vdash \seq{k}, \seq{p} \splitTo K, L \binding \Delta
}
{
  \Gamma \vdash \shapeList{\tau}{k_1, \ldots, k_n}, \patList{p_1, \ldots, p_n} \splitTo \set{\shapeList{\tau}{\seq{m}} \mid \seq{m} \in K}, \set{\shapeList{\tau}{\seq{k}\,'} \mid \seq{k}\,' \in L} \binding \Delta
}
\and
\inferrule*[lab={\ruleName{split-dict-nil}}]
{
  \strut
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{} \splitTo \set{\shapeDict{\tau}{\beta}{N}}, \varnothing \binding \varnothing
}
\and
\inferrule*[lab={\ruleName{split-dict-bound}}]
{
  \beta(w_1) = k \\
  \Gamma \vdash k, p_1 \splitTo K, L \binding \Delta \\
  \Gamma \vdash \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{m}}}{N} \mid m \in K}, \patDict{\bind{w_2}{p_2}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \splitTo K', L' \binding \Delta'
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \splitTo K', \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{k'}}}{N} \mid k' \in L} \cup L' \binding \Delta \override \Delta'
}
\and
\inferrule*[lab={\ruleName{split-dict-unbound}}]
{
  w_1 \notin \dom(\beta) \cup N \\
  \Gamma \vdash \shapesOf(\tau, \varnothing), p_1 \splitTo K, L \binding \Delta \\
  \Gamma \vdash \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{m}}}{N} \mid m \in K}, \patDict{\bind{w_2}{p_2}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \splitTo K', L' \binding \Delta'
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_{n+1}}{p_{n+1}}} \splitTo K', \set{\shapeDict{\tau}{\beta \override \set{\bind{w_1}{k}}}{N} \mid k \in L} \cup L' \cup \set{\shapeDict{\tau}{\beta}{N \cup \set{w_1}}} \binding \Delta \override \Delta'
}
\end{mathpar}
