\begin{align*}
N \covers \tyLiteral{\ell} \quad & \text{iff } \ell \in N \\
N \covers \tyBool \quad & \text{iff } \set{\pyinline{True}, \pyinline{False}} \subseteq N \\
N \covers \tyNone \quad & \text{iff } \pyinline{None} \in N \\
N \covers q \quad & \text{iff } \Gamma \vdash q \subty q' \text{ for some } q' \in N \\
N \covers \tyNever \quad &
\end{align*}

\begin{align*}
\restOf(\tau, N) &= \varnothing \; \text{ if } N \covers \tau, \quad \set{\shapeAny{\tau}{N}} \text{ otherwise} \\
\residual(\tau) &= \textstyle\bigcup \set{\restOf(\sigma, \varnothing) \mid \sigma \text{ in } \membersOf(\tau)}
\end{align*}
