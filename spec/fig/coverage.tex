\shapesOf(\tyNone, N) & = \varnothing &\quad& \text{if } \pyinline{None} \in N \\

\shapesOf(\tyDict{\tau}, N) & = \set{\shapeDict{\tau}{\varnothing}{N}} \\

  \ell \notin N \\

  q \notin N \\

  \Gamma \vdash (\shapeAny{\sigma_1}{\varnothing}, \ldots, \shapeAny{\sigma_n}{\varnothing}), (\fieldMap(C, a)(x_1), \ldots, \fieldMap(C, a)(x_n)) \splitTo K, L

  \Gamma \vdash \shapeAny{\tau}{N}, \patConstr{q}{a} \splitTo \set{\patConstr{q}{\seq{m}} \mid \seq{m} \in K}, \set{\patConstr{q}{\seq{k}} \mid \seq{k} \in L} \cup \shapesOf(\tau, N \cup \set{q})
}
\and
\inferrule*[lab={\ruleName{split-rest-tuple}}]
{
  \Gamma \vdash (\shapeAny{\tau_1}{\varnothing}, \ldots, \shapeAny{\tau_n}{\varnothing}), \seq{p} \splitTo K, L

}
