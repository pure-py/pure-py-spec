\small
\inferrule*[lab={\ruleName{split-literal}}]
{
  \ell \notin N \\
  \tyLiteral{\ell} \subty \tau
}
{
  \Gamma \vdash \shapeAny{\tau}{N}, \ell \splits \set{\ell}, \shapesOf(\tau, N \cup \set{\ell})
}
\and
\inferrule*[lab={\ruleName{split-tuple}}]
{
  \strut
}
{
  \Gamma \vdash \shapeAny{\tyTuple{\tau_1, \ldots, \tau_n}}{\varnothing}, \patTuple{p_1, \ldots, p_n} \splits \set{\patTuple{\seq{k}} \mid \seq{k} \in \shapesOf(\seq{\tau})}, \varnothing
}
\and
\inferrule*[lab={\ruleName{split-list}}]
{
  n \notin N
}
{
  \Gamma \vdash \shapeAny{\tyList{\tau}}{N}, \patList{p_1, \ldots, p_n} \splits \set{\shapeList{\tau}{\seq{k}} \mid \seq{k} \in \shapesOf(\tau, \varnothing)^n}, \shapesOf(\tyList{\tau}, N \cup \set{n})
}
\and
\inferrule*[lab={\ruleName{split-key}}]
{
  w_i \notin \dom(\beta) \cup N \\
  w_j \in \dom(\beta) \quad (\forall j < i)
}
{
  \Gamma \vdash \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} \splits \set{\shapeDict{\tau}{\beta \override \set{\bind{w_i}{k}}}{N} \mid k \in \shapesOf(\tau, \varnothing)}, \set{\shapeDict{\tau}{\beta}{(N \cup \set{w_i})}}
}
\and
\inferrule*[lab={\ruleName{split-class}}]
{
  \Gamma \vdash q : \Type{C} \\
  \nexists D \in N.\ C \subty D \\
  \tau, C \text{ comparable} \\
  E \text{ the lower of } \tau \text{ and } C \\
  \fields(E) = \bind{\seq{x}}{\seq{\sigma}} \\
  N' = \set{h \in N \mid \Gamma \vdash h : E}
}
{
  \Gamma \vdash \shapeAny{\tau}{N}, \patConstr{q}{a} \splits \set{\shapeAny{\patConstr{E}{\seq{k}}}{N'} \mid \seq{k} \in \shapesOf(\seq{\sigma})}, \shapesOf(\tau, N \cup \set{C})
}
\and
\inferrule*[lab={\ruleName{split-subclass}}]
{
  \Gamma \vdash q : \Type{D} \\
  D \subty C \\
  D \neq C \\
  \nexists E \in N.\ D \subty E \\
  \fields(D) = \fields(C) \mathbin{\concat} \bind{\seq{y}}{\seq{\sigma}} \\
  N' = \set{h \in N \mid \Gamma \vdash h : D}
}
{
  \Gamma \vdash \shapeAny{\patConstr{C}{\seq{k}}}{N}, \patConstr{q}{a} \splits \set{\shapeAny{\patConstr{D}{\seq{k} \mathbin{\concat} \seq{k}\,'}}{N'} \mid \seq{k}\,' \in \shapesOf(\seq{\sigma})}, \set{\shapeAny{\patConstr{C}{\seq{k}}}{(N \cup \set{D})}}
}
