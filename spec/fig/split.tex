\begin{mathpar}
\small
\inferrule*[lab={\ruleName{split-literal}}]
{
  \ell \notin N \\
  \tyLiteral{\ell} \subty \tau
}
{
  \Gamma \vdashSplit \shapeAny{\tau}{N}, \ell \matches \set{\ell}, \shapesOf(\tau, N \cup \set{\ell})
}
\and
\inferrule*[lab={\ruleName{split-tuple}}]
{
  \strut
}
{
  \Gamma \vdashSplit \shapeAny{\tyTuple{\tau_1, \ldots, \tau_n}}{\varnothing}, \patTuple{p_1, \ldots, p_n} \matches \set{\patTuple{\seq{k}} \mid \seq{k} \in \shapesOf(\seq{\tau})}, \varnothing
}
\and
\inferrule*[lab={\ruleName{split-list}}]
{
  n \notin N
}
{
  \Gamma \vdashSplit \shapeAny{\tyList{\tau}}{N}, \patList{p_1, \ldots, p_n} \matches \set{\shapeList{\tau}{\seq{k}} \mid \seq{k} \in \shapesOf(\tau, \varnothing)^n}, \shapesOf(\tyList{\tau}, N \cup \set{n})
}
\and
\inferrule*[lab={\ruleName{split-key}}]
{
  w_i \notin \dom(\beta) \cup N \\
  w_j \in \dom(\beta) \quad (\forall j < i)
}
{
  \Gamma \vdashSplit \shapeDict{\tau}{\beta}{N}, \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} \matches \set{\shapeDict{\tau}{\beta \override \set{\bind{w_i}{k}}}{N} \mid k \in \shapesOf(\tau, \varnothing)}, \set{\shapeDict{\tau}{\beta}{(N \cup \set{w_i})}}
}
\and
\inferrule*[lab={\ruleName{split-class}}]
{
  \Gamma \vdash q : \Type{C} \\
  \nexists C' \in N.\ C \subty C' \\
  \tau, C \text{ comparable} \\
  E \text{ the lower of } \tau \text{ and } C \\
  \fields(E) = \bind{\seq{x}}{\seq{\sigma}} \\
  N' = \set{h \in N \mid \Gamma \vdash h : E}
}
{
  \Gamma \vdashSplit \shapeAny{\tau}{N}, \patConstr{q}{a} \matches \set{\shapeAny{\patConstr{E}{\seq{k}}}{N'} \mid \seq{k} \in \shapesOf(\seq{\sigma})}, \shapesOf(\tau, N \cup \set{C})
}
\and
\inferrule*[lab={\ruleName{split-subclass}}]
{
  \Gamma \vdash q : \Type{C'} \\
  C' \subty C \\
  C' \neq C \\
  \nexists C'' \in N.\ C' \subty C'' \\
  \fields(C') = \fields(C) \mathbin{\concat} \bind{\seq{y}}{\seq{\sigma}} \\
  N' = \set{h \in N \mid \Gamma \vdash h : C'}
}
{
  \Gamma \vdashSplit \shapeAny{\patConstr{C}{\seq{k}}}{N}, \patConstr{q}{a} \matches \set{\shapeAny{\patConstr{C'}{\seq{k} \mathbin{\concat} \seq{k}\,'}}{N'} \mid \seq{k}\,' \in \shapesOf(\seq{\sigma})}, \set{\shapeAny{\patConstr{C}{\seq{k}}}{(N \cup \set{C'})}}
}
\end{mathpar}
