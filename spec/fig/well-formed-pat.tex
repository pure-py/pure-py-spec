\begin{mathpar}
\small
\inferrule*[lab={\ruleName{wf-pat-lit}}]
{ \strut }
{ \Gamma \vdash \ell }
\and
\inferrule*[lab={\ruleName{wf-pat-neg-lit}}]
{ \strut }
{ \Gamma \vdash -n }
\and
\inferrule*[lab={\ruleName{wf-pat-var}}]
{ \strut }
{ \Gamma \vdash x }
\and
\inferrule*[lab={\ruleName{wf-pat-wild}}]
{ \strut }
{ \Gamma \vdash \patWild }
\and
\inferrule*[lab={\ruleName{wf-pat-list}}]
{ \Gamma \vdash p_i \quad (\forall i) \\ \text{variables in } \seq{p} \text{ distinct} }
{ \Gamma \vdash \patList{\seq{p}} }
\and
\inferrule*[lab={\ruleName{wf-pat-tuple}}]
{ \Gamma \vdash p_i \quad (\forall i) \\ \text{variables in } \seq{p} \text{ distinct} }
{ \Gamma \vdash \patTuple{\seq{p}} }
\and
\inferrule*[lab={\ruleName{wf-pat-as}}]
{ \Gamma \vdash p \\ x \notin \binds(p) }
{ \Gamma \vdash p\;\pyinline{as}\;x }
\and
\inferrule*[lab={\ruleName{wf-pat-dict}}]
{ \Gamma \vdash p_i \quad (\forall i) \\ w_1, \ldots, w_n \text{ distinct} \\ \text{variables in } \seq{p} \text{ distinct} }
{ \Gamma \vdash \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} }
\and
\inferrule*[lab={\ruleName{wf-pat-constr}}]
{
  \Gamma \vdash q : C \\
   \\
   \\
  \Gamma \vdash \fieldMap(C, a)(x) \quad (\forall x \in \dom(\fields(C))) \\
  \text{variables in } \seq{p} \text{ distinct}
}
{
  \Gamma \vdash \patConstr{q}{a}
}
\and
\inferrule*[lab={\ruleName{wf-cases-cons}}]
{
  \Gamma \vdash p \\
  \Gamma \vdash p' \cons \seq{p}\,' \\
  \Gamma \vdash p_i \not\subsumed p \quad (\forall i)
}
{
  \Gamma \vdash p \cons p' \mathbin{\cons} \seq{p}\,'
}
\end{mathpar}
