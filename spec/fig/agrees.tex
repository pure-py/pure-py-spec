\begin{mathpar}
\small
\inferrule*[lab={\ruleName{agrees-var}}]
{ \strut }
{ \Sigma; \Gamma \vdash x \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-wild}}]
{ \strut }
{ \Sigma; \Gamma \vdash \patWild \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-lit}}]
{ \strut }
{ \Sigma; \Gamma \vdash \ell \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-neg-lit}}]
{ \strut }
{ \Sigma; \Gamma \vdash -n \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-as}}]
{ \Sigma; \Gamma \vdash p \agrees \tau }
{ \Sigma; \Gamma \vdash p\;\pyinline{as}\;x \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-union}}]
{ \Sigma; \Gamma \vdash p \agrees \sigma \\ \Sigma; \Gamma \vdash p \agrees \tau }
{ \Sigma; \Gamma \vdash p \agrees \tyUnion{\sigma}{\tau} }
\and
\inferrule*[lab={\ruleName{agrees-tuple}}]
{ \Sigma; \Gamma \vdash p_i \agrees \tau_i \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patTuple{p_1, \ldots, p_n} \agrees \tyTuple{\tau_1, \ldots, \tau_n} }
\and
\inferrule*[lab={\ruleName{agrees-tuple-other}}]
{ \tau \text{ neither list type, union, nor tuple type of arity } n }
{ \Sigma; \Gamma \vdash \patTuple{p_1, \ldots, p_n} \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-list}}]
{ \Sigma; \Gamma \vdash p_i \agrees \tau \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patList{p_1, \ldots, p_n} \agrees \tyList{\tau} }
\and
\inferrule*[lab={\ruleName{agrees-list-other}}]
{ \tau \text{ neither tuple type, union, nor list type} }
{ \Sigma; \Gamma \vdash \patList{\seq{p}} \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-dict}}]
{ \Sigma; \Gamma \vdash p_i \agrees \tau \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} \agrees \tyDict{\tau} }
\and
\inferrule*[lab={\ruleName{agrees-dict-other}}]
{ \tau \text{ neither dictionary type nor union} }
{ \Sigma; \Gamma \vdash \patDict{\bind{\seq{w}}{\seq{p}}} \agrees \tau }
\and
\inferrule*[lab={\ruleName{agrees-constr}}]
{ \Gamma \vdash q : \Type{C} \\ \Sigma; \Gamma \vdash \fieldMap(\Sigma, C, a)(x) \agrees \sigma \quad (\forall \bind{x}{\sigma} \in \fields(\Sigma, C)) }
{ \Sigma; \Gamma \vdash \patConstr{q}{a} \agrees \tau }
\end{mathpar}
