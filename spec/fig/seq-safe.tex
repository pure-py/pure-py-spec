\flushleft \shadebox{$\Sigma; \Gamma \vdash p \seqSafe \tau$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{seq-safe-var}}]
{ \strut }
{ \Sigma; \Gamma \vdash x \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-wild}}]
{ \strut }
{ \Sigma; \Gamma \vdash \patWild \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-lit}}]
{ \strut }
{ \Sigma; \Gamma \vdash \ell \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-neg-lit}}]
{ \strut }
{ \Sigma; \Gamma \vdash -n \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-as}}]
{ \Sigma; \Gamma \vdash p \seqSafe \tau }
{ \Sigma; \Gamma \vdash p\;\pyinline{as}\;x \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-union}}]
{ \Sigma; \Gamma \vdash p \seqSafe \sigma \\ \Sigma; \Gamma \vdash p \seqSafe \tau }
{ \Sigma; \Gamma \vdash p \seqSafe \tyUnion{\sigma}{\tau} }
\and
\inferrule*[lab={\ruleName{seq-safe-tuple}}]
{ \Sigma; \Gamma \vdash p_i \seqSafe \tau_i \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patTuple{p_1, \ldots, p_n} \seqSafe \tyTuple{\tau_1, \ldots, \tau_n} }
\and
\inferrule*[lab={\ruleName{seq-safe-tuple-other}}]
{ \tau \text{ neither list type, union, nor tuple type of arity } n }
{ \Sigma; \Gamma \vdash \patTuple{p_1, \ldots, p_n} \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-list}}]
{ \Sigma; \Gamma \vdash p_i \seqSafe \tau \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patList{\seq{p}} \seqSafe \tyList{\tau} }
\and
\inferrule*[lab={\ruleName{seq-safe-list-other}}]
{ \tau \text{ neither tuple type, union, nor list type} }
{ \Sigma; \Gamma \vdash \patList{\seq{p}} \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-dict}}]
{ \Sigma; \Gamma \vdash p_i \seqSafe \tau \quad (\forall i) }
{ \Sigma; \Gamma \vdash \patDict{\bind{w_1}{p_1}, \ldots, \bind{w_n}{p_n}} \seqSafe \tyDict{\tau} }
\and
\inferrule*[lab={\ruleName{seq-safe-dict-other}}]
{ \tau \text{ neither dictionary type nor union} }
{ \Sigma; \Gamma \vdash \patDict{\bind{\seq{w}}{\seq{p}}} \seqSafe \tau }
\and
\inferrule*[lab={\ruleName{seq-safe-constr}}]
{ \Gamma \vdash q : \Type{C} \\ \Sigma; \Gamma \vdash \fieldMap(\Sigma, C, a)(x) \seqSafe \sigma \quad (\forall \bind{x}{\sigma} \in \fields(\Sigma, C)) }
{ \Sigma; \Gamma \vdash \patConstr{q}{a} \seqSafe \tau }
\end{mathpar}
