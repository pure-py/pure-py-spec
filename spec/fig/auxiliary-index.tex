\auxrow{$\assignsS(t)$}{set of variables}{variables assigned in $t$, not descending into nested functions}{aux:assigns}
\auxrow{$\binds(p)$, $\binds(\seq{g})$}{set of variables}{variables bound by pattern $p$, or by generators of qualifiers $\seq{g}$}{aux:binds}

\auxgroup{Types}
\auxrow{$\baseType(\ell)$, $\baseType(\tau)$}{type}{base type of literal $\ell$, or of type $\tau$}{aux:base-type}
\auxrow{$\joinTy(\Sigma, \sigma, \tau)$, $\joinTy(\Sigma, \seq{\tau})$}{type}{larger type where comparable, union otherwise}{aux:join}
\auxrow{$\meetTy(\Sigma, \sigma, \tau)$}{type}{greatest type below both}{aux:meet}
\auxrow{$\elemTy(\Sigma, \tau)$}{type}{type of elements drawn from $\tau$ by generator}{aux:elem-type}
\auxrow{$\shapesOf(\Sigma, \tau, N)$, $\shapesOf(\Sigma, \seq{\tau})$}{set of shapes or sequences of shapes}{shapes of $\tau$ once heads in $N$ are excluded; sequences of shapes of $\seq{\tau}$}{aux:shapes}

\auxgroup{Classes and modules}
\auxrow{$\submods(q)$}{context}{stub for each submodule of $q$}{aux:submods}
\auxrow{$\ownFields(\phi)$}{sequence of bindings}{fields declared by field list $\phi$, with their type expressions}{aux:own-fields}
\auxrow{$\fields(\Sigma, C)$}{sequence of bindings}{fields of class $C$ with their types, inherited first}{aux:fields}
\auxrow{$\ancestors(\Sigma, C)$}{set of classes}{$C$ and its base classes}{aux:ancestors}
\auxrow{$\fieldMap(\Sigma, C, \seq{u})$}{map from variables}{argument supplied for each field of $C$; undefined unless arguments cover them exactly}{aux:field-map}
\specauxrow{$\env(\rho, q)$}{environment}{environment of module $q$ as loaded in $\rho$}{aux:env}
\auxrow{$\nameOf(q)$}{string}{string literal naming module $q$}{aux:name}

\auxgroup{Values}
\auxrow{$\eqOp(v, v')$}{Boolean}{structural equality; undefined between unrelated values}{aux:eq}
\specauxrow{$\eqElems(\seq{v}, \seq{v}\,')$}{Boolean}{elementwise equality of value sequences}{aux:eq-elems}
\auxrow{$\memOp(v, v')$}{Boolean}{whether $v'$ is element of $v$}{aux:contains}
\specauxrow{$\memElems(\seq{v}, v)$}{Boolean}{whether $\seq{v}$ has element equal to $v$}{aux:contains-elems}
\specauxrow{$\elems(v)$}{sequence of values}{elements of list, tuple or string}{aux:elems}
\auxrow{$\iterOp(v)$}{sequence of values}{sequence generator draws from $v$; keys if $v$ is dictionary}{aux:iter}
\auxrow{$\getitem(v, v')$}{value or abort}{subscripting; $\errIndex$ for index out of range, $\errKey$ for missing key, $\errType$ otherwise}{aux:getitem}
\auxrow{$\outcome(r)$}{value or abort}{statement result as outcome: returned value, \pyinline{None} if block falls off end, abort unchanged}{aux:outcome}
\auxrow{$\dispatch(\rho, v, \seq{p}, \seq{s})$}{bindings and statement}{case taken by \pyinline{match}}{aux:dispatch}
\specauxrow{$\entries(\delta, \delta')$}{dictionary entries}{$\delta$ extended by $\delta'$ in order, later overriding earlier}{aux:entries}
\specauxrow{$\update(\delta, w, v)$}{dictionary entries}{$\delta$ with $w$ bound to $v$, replacing earlier binding}{aux:update}
