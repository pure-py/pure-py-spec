\flushleft \shadebox{$\Sigma; \Gamma \vdash \psi \synth \tau$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{ty-name}}]
{ \bind{\nu}{\predefName} \in \Gamma }
{ \Sigma; \Gamma \vdash \nu \synth \nu }
\and
\inferrule*[lab={\ruleName{ty-var}}]
{
  \bind{\alpha}{\TypeVar} \in \Gamma
}
{
  \Sigma; \Gamma \vdash \alpha \synth \alpha
}
\and
\inferrule*[lab={\ruleName{ty-class}}]
{
  \Gamma \vdash q : \Type{C} \\
  \Sigma; \Gamma \vdash \seq{\psi} \synth \seq{\tau} \\
  |\seq{\tau}| = |\typeParams{\Sigma}(C)|
}
{
  \Sigma; \Gamma \vdash q[\seq{\psi}] \synth C[\seq{\tau}]
}
\and
\inferrule*[lab={\ruleName{ty-alias}}]
{
  \Gamma \vdash q : \TypeAlias[\Gamma']{q'}{\seq{\chi}}{i} \\
  \chi_i = \exTypeAlias{x[\seq{\alpha}]}{\psi'} \\
  \Sigma; \typeContext{\Gamma'}(q', \seq{\chi}, i) \vdash \psi' \synth \tau \\
  \Sigma; \Gamma \vdash \seq{\psi} \synth \seq{\sigma}
}
{
  \Sigma; \Gamma \vdash q[\seq{\psi}] \synth \subst{\seq{\sigma}}{\seq{\alpha}}{\tau}
}
\and
\inferrule*[lab={\ruleName{ty-literal}}]
{ \bind{\pyinline{Literal}}{\predefName} \in \Gamma }
{ \Sigma; \Gamma \vdash \tyLiteral{\ell} \synth \tyLiteral{\ell} }
\and
\inferrule*[lab={\ruleName{ty-list}}]
{ \bind{\pyinline{list}}{\predefName} \in \Gamma \\ \Sigma; \Gamma \vdash \psi \synth \tau }
{ \Sigma; \Gamma \vdash \tyList{\psi} \synth \tyList{\tau} }
\and
\inferrule*[lab={\ruleName{ty-dict}}]
{ \bind{\pyinline{dict}}{\predefName} \in \Gamma \\ \bind{\pyinline{str}}{\predefName} \in \Gamma \\ \Sigma; \Gamma \vdash \psi \synth \tau }
{ \Sigma; \Gamma \vdash \tyDict{\psi} \synth \tyDict{\tau} }
\and
\inferrule*[lab={\ruleName{ty-tuple}}]
{ \bind{\pyinline{tuple}}{\predefName} \in \Gamma \\ \Sigma; \Gamma \vdash \psi_i \synth \tau_i \quad (\forall i) }
{ \Sigma; \Gamma \vdash \tyTuple{\seq{\psi}} \synth \tyTuple{\seq{\tau}} }
\and
\inferrule*[lab={\ruleName{ty-callable}}]
{ \bind{\pyinline{Callable}}{\predefName} \in \Gamma \\ \Sigma; \Gamma \vdash \psi_i \synth \sigma_i \quad (\forall i) \\ \Sigma; \Gamma \vdash \psi \synth \tau }
{ \Sigma; \Gamma \vdash \tyCallable{\seq{\psi}}{\psi} \synth \tyCallable{\seq{\sigma}}{\tau} }
\and
\inferrule*[lab={\ruleName{ty-union}}]
{ \Sigma; \Gamma \vdash \psi \synth \sigma \\ \Sigma; \Gamma \vdash \psi' \synth \tau }
{ \Sigma; \Gamma \vdash \tyUnion{\psi}{\psi'} \synth \tyUnion{\sigma}{\tau} }
\end{mathpar}
