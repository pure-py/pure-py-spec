\flushleft \shadebox{$\Sigma \vdash h : \tau$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{head-literal}}]
{
  \Sigma \vdash \tyLiteral{\ell} \subty \tau
}
{
  \Sigma \vdash \ell : \tau
}
\and
\inferrule*[lab={\ruleName{head-class}}]
{
  \Sigma \vdash C \subty \tau
}
{
  \Sigma \vdash C : \tau
}
\and
\inferrule*[lab={\ruleName{head-length}}]
{
  n \text{ an integer}
}
{
  \Sigma \vdash n : \tyList{\tau}
}
\end{mathpar}

\vspace{2mm}

\flushleft \shadebox{$\Sigma \vdash t : \seq{\tau}$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{trie-top}}]
{
  \strut
}
{
  \Sigma \vdash \top : \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-bot}}]
{
  \strut
}
{
  \Sigma \vdash \bot : \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-union}}]
{
  \Sigma \vdash t : \tau \cons \seq{\tau} \\
  \Sigma \vdash t' : \sigma \cons \seq{\tau}
}
{
  \Sigma \vdash \tyUnion{t}{t'} : (\tyUnion{\tau}{\sigma}) \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-tuple}}]
{
  \Sigma \vdash t : \seq{\sigma} \concat \seq{\tau}
}
{
  \Sigma \vdash t : \tyTuple{\seq{\sigma}} \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-switch}}]
{
  \Sigma \vdash h_i : \tau \quad
  \Sigma \vdash t_i : \compsOf(\Sigma, \tau, h_i) \concat \seq{\tau} \quad (\forall i) \\
  \tau \text{ neither union, tuple nor dictionary type}
}
{
  \Sigma \vdash \trieSwitch{\bind{h_1}{t_1}, \ldots, \bind{h_n}{t_n}}{d} : \tau \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-dict}}]
{
  \Sigma \vdash t_i : \tau \cons \seq{\tau} \quad (\forall i)
}
{
  \Sigma \vdash \trieDict{\bind{w_1}{t_1}, \ldots, \bind{w_n}{t_n}}{d} : \tyDict{\tau} \cons \seq{\tau}
}
\end{mathpar}

\vspace{2mm}

\begin{defalign}
\compsOf(\Sigma, \tau, \ell) &{}= \varepsilon \\*
\compsOf(\Sigma, \tau, C) &{}= \seq{\sigma} && \text{if } \fields(\Sigma, C) = \seq{\bind{x}{\sigma}} \\*
\compsOf(\Sigma, \tyList{\tau}, n) &{}= \sigma_1, \ldots, \sigma_n && \text{if } \sigma_i = \tau \; (\forall i)
\end{defalign}
