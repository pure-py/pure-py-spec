\flushleft \shadebox{$\Sigma \vdash T : \seq{\tau}$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{trie-top}}]
{
  \strut
}
{
  \Sigma \vdash \top : \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-bot}}]
{
  \strut
}
{
  \Sigma \vdash \bot : \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-union}}]
{
  \Sigma \vdash T : \tau \cons \seq{\tau} \\
  \Sigma \vdash T' : \sigma \cons \seq{\tau}
}
{
  \Sigma \vdash \tyUnion{T}{T'} : (\tyUnion{\tau}{\sigma}) \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-tuple}}]
{
  \Sigma \vdash T : \seq{\sigma} \concat \seq{\tau}
}
{
  \Sigma \vdash T : \tyTuple{\seq{\sigma}} \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-match}}]
{
  \Sigma \vdash \set{\bind{\seq{h}}{\seq{T}}} : \tau \cons \seq{\tau} \\
  \tau \text{ neither union, tuple nor dictionary type}
}
{
  \Sigma \vdash \trieMatch{\set{\bind{\seq{h}}{\seq{T}}}}{U} : \tau \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{trie-dict}}]
{
  \Sigma \vdash T_i : \tau \cons \seq{\tau} \quad (\forall i)
}
{
  \Sigma \vdash \trieDict{\set{\bind{\seq{w}}{\seq{T}}}}{U} : \tyDict{\tau} \cons \seq{\tau}
}
\end{mathpar}

\vspace{2mm}

\flushleft \shadebox{$\Sigma \vdash \set{\bind{\seq{h}}{\seq{T}}} : \seq{\tau}$}
\begin{mathpar}
\small
\inferrule*[lab={\ruleName{entries-empty}}]
{
  \strut
}
{
  \Sigma \vdash \varnothing : \tau \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{entries-literal}}]
{
  \Sigma \vdash \tyLiteral{\ell} \subty \tau \\
  \Sigma \vdash T : \seq{\tau} \\
  \Sigma \vdash \set{\bind{\seq{h}}{\seq{T}}} : \tau \cons \seq{\tau}
}
{
  \Sigma \vdash \set{\bind{\ell}{T}, \bind{\seq{h}}{\seq{T}}} : \tau \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{entries-class}}]
{
  \Sigma \vdash C \subty \tau \\
  \fields(\Sigma, C) = \bind{\seq{x}}{\seq{\sigma}} \\
  \Sigma \vdash T : \seq{\sigma} \concat \seq{\tau} \\
  \Sigma \vdash \set{\bind{\seq{h}}{\seq{T}}} : \tau \cons \seq{\tau}
}
{
  \Sigma \vdash \set{\bind{C}{T}, \bind{\seq{h}}{\seq{T}}} : \tau \cons \seq{\tau}
}
\and
\inferrule*[lab={\ruleName{entries-length}}]
{
  \sigma_i = \tau \quad (\forall i \leq n) \\
  \Sigma \vdash T : \sigma_1, \ldots, \sigma_n \concat \seq{\tau} \\
  \Sigma \vdash \set{\bind{\seq{h}}{\seq{T}}} : \tyList{\tau} \cons \seq{\tau}
}
{
  \Sigma \vdash \set{\bind{n}{T}, \bind{\seq{h}}{\seq{T}}} : \tyList{\tau} \cons \seq{\tau}
}
\end{mathpar}
