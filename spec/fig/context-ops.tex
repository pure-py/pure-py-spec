\flushleft \shadebox{$\Gamma \override \Delta$ (override)}
\begin{defalign}
(\Gamma \override \Delta)(x) &= \Gamma(x) \override \Delta(x) \\
\bot \override \theta &= \theta \override \bot = \theta \\
\theta \override \theta' &= \theta' \quad (\text{otherwise})
\end{defalign}
\flushleft \shadebox{$\Gamma \merge \Delta$ (merge)}
\begin{defalign}
(\Gamma \merge \Delta)(x) &= \Gamma(x) \merge \Delta(x) \\
\bot \merge \bot &= \bot \\
a \merge \bot = \bot \merge a &= \statusNotDefAssigned \\
a \merge a' &= a \meet a' \quad (a, a' \in \B)
\end{defalign}
\flushleft \shadebox{$\Gamma \extend \Delta$ (extension)}
\begin{defalign}
(\Gamma \extend \Delta)(x) &= \Gamma(x) \extend \Delta(x) \\
\modLoaded{q}{\Gamma} \extend \modLoaded{q}{\Gamma'} &= \modLoaded{q}{\Gamma \extend \Gamma'} \\
\modStub{q} \extend \modLoaded{q}{\Gamma} = \modLoaded{q}{\Gamma} \extend \modStub{q} &= \modLoaded{q}{\Gamma} \\
\bot \extend \theta = \theta \extend \bot &= \theta \\
\theta \extend \theta' &= \theta' \quad (\text{otherwise})
\end{defalign}
