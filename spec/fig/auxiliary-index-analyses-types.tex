% Rows of the auxiliary-functions index; the auxindex environment supplies the table.
\auxgroup{Syntactic analyses}
\auxrow{$\assignsS(t)$}{set of variables}{variables declared or assigned in $t$, not descending into nested functions}{aux:assigns}
\auxrow{$\declaresS(t)$, $\declaresS(\seq{\iota})$}{sequence of variables}{names declared in $t$, or bound by imports $\seq{\iota}$, in order and with repeats, not descending into nested functions}{aux:declares}
\auxrow{$\typeEntry{\Gamma}(q, \seq{\chi}, i)$, $\typeEntry{\Gamma}(q, \seq{\chi})$}{context entry, or sequence of context entries}{context entry associated with definition $\chi_i$ of mutual type region $\seq{\chi}$ of module $q$ under $\Gamma$, or with each definition of $\seq{\chi}$}{aux:type-entry}
\auxrow{$\typeContext{\Gamma}(q, \seq{\chi}, i)$}{context}{context resolving type expressions of definition $\chi_i$ of mutual type region $\seq{\chi}$ of module $q$ under $\Gamma$}{aux:type-context}
\auxrow{$\classTable{\Gamma}(q, \seq{\chi}, i)$, $\classTable{\Gamma}(q, \seq{\chi})$}{class table}{class table contributed by definition $\chi_i$, or by all definitions, of mutual type region $\seq{\chi}$ of module $q$ under $\Gamma$}{aux:class-table}
\auxrow{$\scopeS(Y, s)$}{context}{each variable declared or assigned in $s$ unbound; defined iff names declared in $s$, names in $Y$ and pattern variables of $s$ pairwise distinct}{aux:scope}
\auxrow{$\binds(p)$, $\binds(\seq{g})$, $\binds(s)$}{set of variables}{variables bound by pattern $p$, by generators of qualifiers $\seq{g}$, or by patterns within statement $s$}{aux:binds}
\auxrow{$\captures(e)$}{set of variables}{variables of enclosing scope captured by lambdas in $e$}{aux:captures-e}
\auxrow{$\captures(\seq{g})$}{set of variables}{variables of enclosing scope captured by lambdas in qualifiers $\seq{g}$}{aux:captures}
\specauxrow{$\fv(e)$, $\fv(\seq{e})$, $\fv(\tau)$}{set of variables}{free variables of expression $e$, or of each expression in $\seq{e}$; type variables of type $\tau$}{aux:fv}
\noalign{\vskip 2mm}
\auxgroup{Types}
\auxrow{$\baseType(\ell)$, $\baseType(\tau)$}{type}{base type of literal $\ell$, or of type $\tau$}{aux:base-type}
\auxrow{$\joinTy{\Sigma}{\sigma}{\tau}$, $\bigJoinTy{\Sigma}{\seq{\tau}}$}{type}{larger type if comparable, and union otherwise}{aux:join}
\auxrow{$\meetTy{\Sigma}{\sigma}{\tau}$}{type}{greatest type below both}{aux:meet}
\auxrow{$\subterms(\tau)$, $\subterms(\seq{\tau})$}{set of types}{subterms of $\tau$, or of each type in $\seq{\tau}$, other than unions}{aux:subterms}
\auxrow{$\typeArgs{\Sigma}(\seq{\alpha}, \sigma, \tau)$, $\typeArgs{\Sigma}(\seq{\alpha}, \seq{\sigma}, \seq{\tau})$}{context binding type variables at types}{type arguments from one-way unification of $\sigma$ with $\tau$, or of $\seq{\sigma}$ with $\seq{\tau}$}{aux:type-args}
\auxrow{$\elemTy{\Sigma}(\tau)$}{type}{type of elements drawn from $\tau$ by generator}{aux:elem-type}
\auxrow{$\shapes{\Sigma}(\tau, H)$, $\shapes{\Sigma}(\seq{\tau})$}{ordered set of shapes or of shape sequences}{shapes of $\tau$ once heads in $H$ are excluded; sequences of shapes of $\seq{\tau}$}{aux:shapes}
\auxrow{$\shapeTy(k)$}{type}{least type of shape $k$}{aux:shape-type}
\auxrow{$\matchedAs{\Sigma}(\Gamma, \tau, p)$}{type}{type $\tau$ is matched as by pattern $p$}{aux:matched-as}
\auxrow{$\remaining{\Sigma}(\Gamma, \tau, p)$}{type}{type remaining from $\tau$ if pattern $p$ fails to match}{aux:remaining}
\auxrow{$\resolveOp{\Sigma}(\odot, \sigma_1, \sigma_2)$, $\resolveOp{\Sigma}(\ominus, \sigma)$}{type}{result type of most specific overload at the operand types}{aux:resolve-op}
